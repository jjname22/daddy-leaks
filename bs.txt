    local function IsTeammate(player)
        if not player then return true end
        if player == LocalPlayer then return true end

        local success1, mySquad = pcall(function() return LocalPlayer:GetAttribute("SquadName") end)
        local success2, theirSquad = pcall(function() return player:GetAttribute("SquadName") end)
        if success1 and success2 and mySquad and theirSquad and mySquad == theirSquad then
            return true
        end

        local successRank1, myRank = pcall(function() return LocalPlayer:GetRankInGroup(3441839) end)
        local successRank2, theirRank = pcall(function() return player:GetRankInGroup(3441839) end)
        if successRank1 and successRank2 and myRank and theirRank and myRank > 0 and theirRank > 0 then
            return true
        end
        
        return false
    end

    local function getVehicleHealth(targetPart)
        local x, y, z = math.floor(targetPart.Position.X), math.floor(targetPart.Position.Y), math.floor(targetPart.Position.Z)
        local vehicleKey = string.format('vehicle_%d_%d_%d', x, y, z)

        if vehicleESPState.WorldModelCache[vehicleKey] then
            local cachedWorldModel = vehicleESPState.WorldModelCache[vehicleKey]
            if cachedWorldModel and cachedWorldModel.Parent then
                local states = cachedWorldModel:FindFirstChild('States')
                if states then
                    local health = states:GetAttribute('Health')
                    if health and health > 0 then
                        return health
                    end
                end
            else
                vehicleESPState.WorldModelCache[vehicleKey] = nil
            end
        end

        local closestWorldModel = nil
        local closestDistance = math.huge

        for _, obj in pairs(Workspace:GetChildren()) do
            if obj.Name == 'WorldModel' and obj:IsA('Model') then
                local states = obj:FindFirstChild('States')
                if states then
                    local health = states:GetAttribute('Health')
                    if health and health > 0 then
                        local cf = obj:GetBoundingBox()
                        if cf then
                            local distance = (cf.Position - targetPart.Position).Magnitude
                            if distance < closestDistance and distance < 150 then
                                closestDistance = distance
                                closestWorldModel = obj
                            end
                        end
                    end
                end
            end
        end

        if closestWorldModel then
            vehicleESPState.WorldModelCache[vehicleKey] = closestWorldModel
            local states = closestWorldModel:FindFirstChild('States')
            if states then
                local health = states:GetAttribute('Health')
                if health and health > 0 then
                    return health
                end
            end
        end

        return nil
    end

                            local charObj = data.DistChars[i]
                            if not charObj then
                                charObj = Drawing.new("Text")
                                charObj.Center = false
                                charObj.Outline = true
                                charObj.OutlineColor = Color3.fromRGB(15, 15, 15)
                                charObj.Font = 2
                                data.DistChars[i] = charObj
                            end
                            charObj.Text = distStr:sub(i, i)
                            charObj.Size = flags.DistanceSize
                            charObj.Transparency = 1 - flags.DistanceTransparency
                            charObj.Color = GetWaveColor(flags.DistanceColor, flags.DistanceColor2, i, 5, 0.4)
                            charObj.Position = Vector2.new(startX + ((i - 1) * charWidth), botVec.Y + 4)
                            charObj.Visible = true
                        end
                        for i = #distStr + 1, #data.DistChars do
                            if data.DistChars[i] then data.DistChars[i].Visible = false end
                        end
                    else
                        for _, cObj in pairs(data.DistChars) do cObj.Visible = false end
                        data.Dist.Position = Vector2.new(botVec.X, botVec.Y + 4)
                        data.Dist.Text = distStr
                        data.Dist.Color = flags.DistanceColor
                        data.Dist.Size = flags.DistanceSize
                        data.Dist.Transparency = 1 - flags.DistanceTransparency
                        data.Dist.Visible = true
                    end
                else
                    data.Dist.Visible = false
                    for _, cObj in pairs(data.DistChars) do cObj.Visible = false end
                end

                if flags.SkeletonEnabled then
                    for i, bone in ipairs(SkeletonBones) do
                        local partA = model:FindFirstChild(bone[1])
                        local partB = model:FindFirstChild(bone[2])
                        local line = data.Skeleton[i]

                        if partA and partB then
                            local posA, visA = Camera:WorldToViewportPoint(partA.Position)
                            local posB, visB = Camera:WorldToViewportPoint(partB.Position)

                            if visA and visB then
                                line.From = Vector2.new(posA.X, posA.Y)
                                line.To = Vector2.new(posB.X, posB.Y)
                                line.Color = flags.SkeletonColor
                                line.Thickness = flags.SkeletonThickness
                                line.Transparency = 1 - flags.SkeletonTransparency
                                line.Visible = true
                            else
                                line.Visible = false
                            end
                        else
                            line.Visible = false
                        end
                    end
                else
                    for _, line in pairs(data.Skeleton) do line.Visible = false end
                end
            end
        end
    end)


    local vehicleScanTimer = 0
    local vehicleScanInterval = 0.75

    local function scanVehicles()
        if not flags.VehicleESPEnabled then
            return
        end

        for _, vehicle in ipairs(Workspace:GetChildren()) do
            if vehicle:IsA("Model") then
                local chassis = vehicle:FindFirstChild("Chassis")

                if chassis and chassis:IsA("BasePart") then
                    if not vehicleESPState.Active[chassis] then
                        createVehicleBillboard(chassis)
                    end
                end
            end
        end
    end

    local function hideVehicleData(data)
        if not data then
            return
        end

        if data.billboard then
            data.billboard.Enabled = false
        end

        if data.nameLabel then
            data.nameLabel.Visible = false
        end

        if data.distLabel then
            data.distLabel.Visible = false
        end

        if data.healthLabel then
            data.healthLabel.Visible = false
        end

        for _, charObj in pairs(data.NameChars or {}) do
            charObj.Visible = false
        end

        for _, charObj in pairs(data.DistChars or {}) do
            charObj.Visible = false
        end
    end

    local function cleanupVehicles()
        for chassis, data in pairs(vehicleESPState.Active) do
            if not chassis or not chassis.Parent then

                if data then
                    if data.billboard then
                        pcall(function()
                            data.billboard:Destroy()
                        end)
                    end

                    for _, charObj in pairs(data.NameChars or {}) do
                        pcall(function()
                            charObj:Remove()
                        end)
                    end

                    for _, charObj in pairs(data.DistChars or {}) do
                        pcall(function()
                            charObj:Remove()
                        end)
                    end
                end

                vehicleESPState.Active[chassis] = nil
            end
        end

        for key, worldModel in pairs(vehicleESPState.WorldModelCache) do
            if not worldModel or not worldModel.Parent then
                vehicleESPState.WorldModelCache[key] = nil
            end
        end
    end

    local function updateVehicleESP()
        Camera = Workspace.CurrentCamera or Camera

        local character = LocalPlayer.Character
        local root = character and character:FindFirstChild("HumanoidRootPart")

        local origin

        if root then
            origin = root.Position
        else
            origin = Camera.CFrame.Position
        end

        for targetPart, data in pairs(vehicleESPState.Active) do

            if not targetPart or not targetPart.Parent then
                hideVehicleData(data)
                continue
            end

            local distance = (targetPart.Position - origin).Magnitude

            if not flags.VehicleESPEnabled or distance > 5000 then
                hideVehicleData(data)
                continue
            end

            local displayDist = distance / 3.5461
            local hasContent = false
            
            local viewportPosition, onScreen = Camera:WorldToViewportPoint(targetPart.Position + Vector3.new(0, 3, 0))
            
            if not onScreen then
                hideVehicleData(data)
                continue
            end

            local currentYOffset = 0
            local defaultSize = 14

            if flags.VehicleNameEnabled then
                local textStr = "[Vehicle]"

                if flags.VehicleNameAnimated then
                    data.nameLabel.Visible = false

                    local charWidth = defaultSize * 0.55
                    local totalWidth = #textStr * charWidth
                    local startX = viewportPosition.X - (totalWidth / 2)

                    for i = 1, #textStr do
                        local charObj = data.NameChars[i]

                        if not charObj then
                            charObj = Drawing.new("Text")
                            charObj.Center = false
                            charObj.Outline = true
                            charObj.OutlineColor = Color3.fromRGB(15, 15, 15)
                            charObj.Font = 2
                            data.NameChars[i] = charObj
                        end

                        charObj.Text = textStr:sub(i, i)
                        charObj.Size = defaultSize
                        charObj.Transparency = 1
                        charObj.Color = GetWaveColor(
                            flags.VehicleNameColor,
                            flags.VehicleNameColor2,
                            i,
                            5,
                            0.4
                        )

                        charObj.Position = Vector2.new(
                            startX + ((i - 1) * charWidth),
                            viewportPosition.Y + currentYOffset
                        )

                        charObj.Visible = true
                    end

                    for i = #textStr + 1, #data.NameChars do
                        if data.NameChars[i] then
                            data.NameChars[i].Visible = false
                        end
                    end
                else
                    for _, charObj in pairs(data.NameChars) do
                        charObj.Visible = false
                    end

                    data.nameLabel.Text = textStr
                    data.nameLabel.TextColor3 = flags.VehicleNameColor
                    data.nameLabel.TextSize = defaultSize
                    data.nameLabel.Position = UDim2.new(0, viewportPosition.X, 0, viewportPosition.Y + currentYOffset)
                    data.nameLabel.Visible = true
                end

                currentYOffset = currentYOffset + 15
                hasContent = true
            else
                data.nameLabel.Visible = false
                for _, charObj in pairs(data.NameChars) do
                    charObj.Visible = false
                end
            end

            if flags.VehicleDistanceEnabled then
                local distStr = string.format("[%d M]", math.floor(displayDist))

                if flags.VehicleDistanceAnimated then
                    data.distLabel.Visible = false

                    local charWidth = defaultSize * 0.55
                    local totalWidth = #distStr * charWidth
                    local startX = viewportPosition.X - (totalWidth / 2)

                    for i = 1, #distStr do
                        local charObj = data.DistChars[i]

                        if not charObj then
                            charObj = Drawing.new("Text")
                            charObj.Center = false
                            charObj.Outline = true
                            charObj.OutlineColor = Color3.fromRGB(15, 15, 15)
                            charObj.Font = 2
                            data.DistChars[i] = charObj
                        end

                        charObj.Text = distStr:sub(i, i)
                        charObj.Size = defaultSize
                        charObj.Transparency = 1
                        charObj.Color = GetWaveColor(
                            flags.VehicleDistanceColor,
                            flags.VehicleDistanceColor2,
                            i,
                            5,
                            0.4
                        )

                        charObj.Position = Vector2.new(
                            startX + ((i - 1) * charWidth),
                            viewportPosition.Y + currentYOffset
                        )

                        charObj.Visible = true
                    end

                    for i = #distStr + 1, #data.DistChars do
                        if data.DistChars[i] then
                            data.DistChars[i].Visible = false
                        end
                    end
                else
                    for _, charObj in pairs(data.DistChars) do
                        charObj.Visible = false
                    end

                    data.distLabel.Text = distStr
                    data.distLabel.TextColor3 = flags.VehicleDistanceColor
                    data.distLabel.TextSize = defaultSize
                    data.distLabel.Position = UDim2.new(0, viewportPosition.X, 0, viewportPosition.Y + currentYOffset)
                    data.distLabel.Visible = true
                end

                currentYOffset = currentYOffset + 15
                hasContent = true
            else
                data.distLabel.Visible = false
                for _, charObj in pairs(data.DistChars) do
                    charObj.Visible = false
                end
            end

            if flags.VehicleHealthEnabled then
                local health = getVehicleHealth(targetPart)

                if health then
                    data.healthLabel.Text = string.format("HP: %.0f", health)
                    data.healthLabel.TextColor3 = Color3.fromRGB(0, 255, 0)
                else
                    data.healthLabel.Text = "HP: N/A"
                    data.healthLabel.TextColor3 = Color3.fromRGB(255, 0, 0)
                end

                data.healthLabel.TextSize = 13
                data.healthLabel.Position = UDim2.new(0, viewportPosition.X, 0, viewportPosition.Y + currentYOffset)
                data.healthLabel.Visible = true

                hasContent = true
            else
                data.healthLabel.Visible = false
            end

            data.billboard.Enabled = hasContent
        end
    end

    RunService.RenderStepped:Connect(function()
        if flags.VehicleESPEnabled then
            vehicleScanTimer = vehicleScanTimer + 1 / 60

            if vehicleScanTimer >= vehicleScanInterval then
                scanVehicles()
                cleanupVehicles()
                vehicleScanTimer = 0
            end

            updateVehicleESP()
        else
            for _, data in pairs(vehicleESPState.Active) do
                hideVehicleData(data)
            end
        end
    end)
end

do
    Players = Players or game:GetService("Players")
    LocalPlayer = Players.LocalPlayer
    Camera = workspace.CurrentCamera
    RunService = game:GetService('RunService')
    
    graveESP = graveESP or {
        Enabled = false,
        ShowName = false,
        ShowDistance = false,
        Animation = true,
        Color1 = Color3.fromRGB(255, 255, 255),
        Color2 = Color3.fromRGB(0, 150, 255), 
        Size = 14,
        RenderDistance = 10000,
        Active = {},
        Font = 2,
    }
    
    local Functions = {}
    do
        function Functions:FadeOutOnDist(drawingsList, distance, maxDistance)
            local transparency = math.max(0.1, 1 - (distance / maxDistance))
            for _, obj in ipairs(drawingsList) do
                obj.Transparency = transparency
            end
        end
    end

    local function getPrimaryPart(model)
        if model.PrimaryPart then
            return model.PrimaryPart
        end
        for _, part in pairs(model:GetDescendants()) do
            if part:IsA("BasePart") then
                return part
            end
        end
        return nil
    end

    local function getGravePlayerName(model)
        local nameLower = model.Name:lower()
        if nameLower:find("zombie") or nameLower:find("npc") or nameLower:find("dummy") then
            return nil
        end

        if model:GetAttribute("OwnerName") then
            return model:GetAttribute("OwnerName")
        end
        if model:GetAttribute("DisplayName") then
            return model:GetAttribute("DisplayName")
        end
        
        for _, descendant in ipairs(model:GetDescendants()) do
            local descNameLower = descendant.Name:lower()
            if descNameLower:find("zombie") or descNameLower:find("npc") then
                return nil
            end

            if descendant:GetAttribute("DisplayName") then
                return descendant:GetAttribute("DisplayName")
            end
            if descendant:GetAttribute("OwnerName") then
                return descendant:GetAttribute("OwnerName")
            end
            if descendant:IsA("StringValue") and (descendant.Name == "Owner" or descendant.Name == "Player") then
                return descendant.Value
            end
        end

        for _, player in ipairs(Players:GetPlayers()) do
            if model.Name == player.Name or model.Name == player.DisplayName then
                return player.Name
            end
        end

        if model.Name == "Default" or model.Name == "Model" or model.Name == "PlayerGrave" then
            return nil
        end

        return model.Name
    end

    local function cleanupGraveData(data)
        if data and data.charDrawings then
            for _, obj in ipairs(data.charDrawings) do
                pcall(function() obj:Remove() end)
            end
            data.charDrawings = {}
        end
    end

    local function createGraveESP(model)
        if graveESP.Active[model] then return end
        
        local playerName = getGravePlayerName(model)
        if not playerName then return end

        local part = getPrimaryPart(model)
        if not part then return end

        graveESP.Active[model] = {
            model = model,
            part = part,
            playerName = playerName,
            charDrawings = {},
            lastText = ""
        }
    end
    
    local function scanGraves()
        if not graveESP.Enabled then return end
        
        local folder = workspace:FindFirstChild("WorldModels") and workspace.WorldModels:FindFirstChild("PlayerGraves")
        if folder then
            local children = folder:GetChildren()
            for i = 1, #children do
                local model = children[i]
                if model:IsA("Model") and not graveESP.Active[model] then
                    createGraveESP(model)
                end
            end
        end
        
        local children = workspace:GetChildren()
        for i = 1, #children do
            local obj = children[i]
            if obj:IsA("Model") and (obj.Name:find("Grave") or obj:FindFirstChild("PlayerGrave")) and not graveESP.Active[obj] then
                createGraveESP(obj)
            end
        end
    end
    
    local function updateGraveESP()
        Camera = workspace.CurrentCamera or Camera
        local char = LocalPlayer.Character
        local InitializerPos = (char and char.PrimaryPart and char.PrimaryPart.Position) or Camera.CFrame.Position
        
        for model, data in pairs(graveESP.Active) do
            local part = data.part
            
            if not model or not model.Parent or not part or not part.Parent or not model:IsDescendantOf(workspace) then
                cleanupGraveData(data)
                graveESP.Active[model] = nil
            else
                local dist = (part.Position - InitializerPos).Magnitude
                local displayDist = dist / 2.56
                
                if graveESP.Enabled and dist <= graveESP.RenderDistance then
                    local targetWorldPos = part.Position + Vector3.new(0, 2.5, 0)
                    local pos, onScreen = Camera:WorldToViewportPoint(targetWorldPos)
                    
                    if onScreen then
                        local text = ''
                        if graveESP.ShowName or graveESP.ShowDistance then
                            if graveESP.ShowName then
                                text = data.playerName
                            end
                            if graveESP.ShowDistance then
                                local distText = string.format('[%.0fm]', displayDist)
                                text = (text ~= '' and (text .. ' ' .. distText)) or distText
                            end
                        end
                        
                        if text == '' then
                            for _, obj in ipairs(data.charDrawings) do obj.Visible = false end
                        else
                            if text ~= data.lastText then
                                cleanupGraveData(data)
                                data.lastText = text
                                for i = 1, #text do
                                    local charObj = Drawing.new("Text")
                                    charObj.Visible = false
                                    charObj.Center = true 
                                    charObj.Outline = true
                                    charObj.OutlineColor = Color3.fromRGB(0, 0, 0)
                                    charObj.Size = graveESP.Size
                                    charObj.Font = graveESP.Font
                                    table.insert(data.charDrawings, charObj)
                                end
                            end
                            
                            local charSpacing = graveESP.Size * 0.55
                            local totalWidth = #text * charSpacing
                            local startX = pos.X - (totalWidth / 2) + (charSpacing / 2)
                            local startY = pos.Y
                            
                            local timeFactor = os.clock() * 4
                            
                            for i, charObj in ipairs(data.charDrawings) do
                                charObj.Visible = graveESP.Enabled
                                charObj.Size = graveESP.Size
                                charObj.Font = graveESP.Font
                                charObj.Text = text:sub(i, i)
                                charObj.Position = Vector2.new(startX + ((i - 1) * charSpacing), startY)
                                
                                local wave = 0.5
                                if graveESP.Animation then
                                    wave = (math.sin(timeFactor - (i * 0.35)) + 1) / 2
                                end
                                charObj.Color = graveESP.Color1:Lerp(graveESP.Color2, wave)
                            end
                            
                            Functions:FadeOutOnDist(data.charDrawings, dist, graveESP.RenderDistance)
                        end
                    else
                        for _, obj in ipairs(data.charDrawings) do obj.Visible = false end
                    end
                else
                    for _, obj in ipairs(data.charDrawings) do obj.Visible = false end
                end
            end
        end
    end
    
    local graveScanInterval = 1.0
    local lastGraveScan = 0
    
    RunService.Heartbeat:Connect(function()
        local now = os.clock()
        if graveESP.Enabled and (graveESP.ShowName or graveESP.ShowDistance) then
            if now - lastGraveScan >= graveScanInterval then
                scanGraves()
                lastGraveScan = now
            end
            updateGraveESP()
        else
            for _, data in pairs(graveESP.Active) do
                for _, obj in ipairs(data.charDrawings or {}) do
                    obj.Visible = false
                end
            end
        end
    end)

    GroupBoxes.Grave:AddToggle("GraveNameToggle", {
        Text = 'Grave Name',    
        Default = false,    
        Callback = function(state)
            graveESP.ShowName = state
            graveESP.Enabled = graveESP.ShowName or graveESP.ShowDistance
            if graveESP.Enabled then
                task.defer(scanGraves)
            else
                for _, data in pairs(graveESP.Active) do
                    cleanupGraveData(data)
                end
            end
        end,    
    }):AddColorPicker("GraveESPColor1", {
        Title = 'Grave Color 1',    
        Default = graveESP.Color1,    
        Callback = function(value)
            graveESP.Color1 = value
        end,    
    }):AddColorPicker("GraveESPColor2", {
        Title = 'Grave Color 2',

    local function predict_priv9(origin, pos, vel, speed, drop)
        local dist = (origin - pos).Magnitude
        local t = dist / speed
        return pos + (vel * t) + Vector3.yAxis * (drop * t * t)
    end

    local function full_prediction_silent(target_position, target_collider)
        if not target_position then return end

        local currentgun = get_current_gun(LocalPlayer)
        local gun = aftermath.gundata:FindFirstChild(currentgun)
        local stats = gun and gun:FindFirstChild("Stats")
        local bullet_settings = stats and stats:FindFirstChild("BulletSettings")

        local proj_speed, proj_drop

        if bullet_settings then
            local bullet_speed = bullet_settings:FindFirstChild("BulletSpeed")
            local bullet_gravity = bullet_settings:FindFirstChild("BulletGravity")
            proj_speed = tonumber(bullet_speed and bullet_speed.Value) or tonumber(aftermath.sv_config.sv_default_bullet_speed.Value) or 1500
            proj_drop = tonumber(bullet_gravity and bullet_gravity.Value) or tonumber(aftermath.sv_config.sv_default_bullet_gravity.Value) or 0
        else
            proj_speed = tonumber(aftermath.sv_config.sv_default_bullet_speed.Value) or 1500
            proj_drop = tonumber(aftermath.sv_config.sv_default_bullet_gravity.Value) or 0
        end

        local campos = Camera.CFrame.Position
        local velocity = (target_collider and target_collider.AssemblyLinearVelocity) or Vector3.zero

        if flags.ResolveY then velocity = Vector3.new(velocity.X, 0, velocity.Z) end
        if flags.instant_bullet then velocity = Vector3.zero end

        return predict_priv9(campos, target_position, velocity, proj_speed, proj_drop)
    end

    local function get_closest_target(fov_size, aimpart)
        local ermm_part, plr_instance, collider
        local maximum_distance = fov_size
        local mousepos = UserInputService:GetMouseLocation()

        local list = get_entity_list()
        if not list then return nil, nil, nil end

        for userid, v in pairs(list) do
            local player = v.Player
            if not (player and player ~= LocalPlayer) then continue end

            if flags.TeamCheck and IsTeammate(player) then continue end

            local root = v.RootPart
            local worldmodel = v.WorldModel
            local character = v.Character

            if not (root and worldmodel and character) then continue end

            local targetPart = nil
            
            if aimpart == "Head" then
                for _, obj in ipairs(character:GetDescendants()) do
                    if obj:GetAttribute("StretchedHead") then
                        targetPart = obj
                        break
                    end
                end
            end
            
            if not targetPart then
                targetPart = worldmodel:FindFirstChild(aimpart)
            end

            if not targetPart then continue end

            local position, onscreen = Camera:WorldToViewportPoint(targetPart.Position)
            local distance = (Vector2.new(position.X, position.Y) - mousepos).Magnitude

            if onscreen and distance <= maximum_distance then
                plr_instance = player
                ermm_part = targetPart
                collider = root
                maximum_distance = distance
            end
        end

        return ermm_part, plr_instance, collider
    end

            if TimeOffset - 0.255 > 0 then
                NextInstant = os.clock() + (TimeOffset - 0.26)
                task.wait(TimeOffset - 0.26)
            else
                NextInstant = os.clock() + TimeOffset
            end
            debug.setstack(3, 1, stack)
            return oldBufferCreate(size, ...)
        end

        if flags.AimType ~= "Silent Aim" then
            return oldBufferCreate(size, ...)
        end

        local part, player, collider = get_closest_target(flags.FovSize or 120, flags.AimPart)
        if not part then
            return oldBufferCreate(size, ...)
        end

        local pred = full_prediction_silent(part.Position, collider) or part.Position
        
        -- Force override origin to the camera position to bypass back-facing anti-aim model distortion
        local overrideOrigin = Camera.CFrame.Position
        local direction = (pred - overrideOrigin).Unit

        -- Apply spread modifications if enabled
        if flags.NoSpread and stack[48] and stack[22] then
            local rng = Random.new(stack[48] + 1)
            local spread = Vector3.new(
                rng:NextNumber() - rng:NextNumber(),
                rng:NextNumber() - rng:NextNumber(),
                rng:NextNumber() - rng:NextNumber()
            ) / stack[22]
            direction = (direction - spread).Unit
        end

        local cf = CFrame.lookAt(overrideOrigin, pred)
        local pitch2, yaw2 = cf:ToEulerAnglesYXZ()
        local r00, r01, r02, r10, r11, r12, r20, r21, r22 = cf:GetComponents()

        -- Deep override of gun stack matrices and origins to force bullet redirection
        stack[32] = cf                    -- Gun CFrame / Origin Matrix
        stack[33] = direction             -- Fire Direction Vector
        stack[34] = direction             -- Secondary Ray Vector
        stack[36] = pitch2                -- Pitch
        stack[37] = yaw2                  -- Yaw
        stack[38] = cf                    -- Transform Matrix Slot A
        stack[39] = cf                    -- Transform Matrix Slot B
        stack[44] = cf                    -- Muzzle CFrame Matrix
        stack[45] = direction             -- Muzzle LookVector
        stack[46] = direction             

        if typeof(stack[30]) == "Vector3" then
            stack[30] = overrideOrigin
        end
        if typeof(stack[31]) == "Vector3" then
            stack[31] = overrideOrigin
        end

        debug.setstack(3, 1, stack)
        return oldBufferCreate(size, ...)
    end))

    local function IsVisibleAtPosition(testPosition, OriginPos, extraIgnore)
        local camera = workspace.CurrentCamera
        OriginPos = OriginPos or (camera and camera.CFrame.Position) or Vector3.new()
        
        table.clear(cachedIgnore)
        if LocalPlayer and LocalPlayer.Character then
            table.insert(cachedIgnore, LocalPlayer.Character)
        end
        if camera then table.insert(cachedIgnore, camera) end
        if typeof(extraIgnore) == 'table' then
            for _, v in ipairs(extraIgnore) do 
                if typeof(v) == 'Instance' then 
                    table.insert(cachedIgnore, v) 
                end 
            end
        elseif typeof(extraIgnore) == 'Instance' then 
            table.insert(cachedIgnore, extraIgnore) 
        end
        
        cachedRayParams.FilterDescendantsInstances = cachedIgnore
        
        local direction = testPosition - OriginPos
        if direction.Magnitude == 0 then return true, nil end
        
        local result = workspace:Raycast(OriginPos, direction, cachedRayParams)
        if not result then return true, nil end
        return false, result
    end

    local function locateTargettedUser()

        local midpoint = CurrentViewport.ViewportSize * 0.5

            if worldEntity and worldEntity.Parent then
                local entityName = fetchPlayerNameFromWorld(worldEntity)
                if entityName and entityName ~= ClientPlayer.Name then
                    local userObj = CorePlayers:FindFirstChild(entityName)
                    if userObj and flags.teamcheck_inventory and IsTeammate(userObj) then
                        continue
                    end

                    local originPart = worldEntity:FindFirstChild("HumanoidRootPart")
                    if originPart then
                        local screenCoords, visibleCheck = CurrentViewport:WorldToViewportPoint(originPart.Position)
                        if visibleCheck then
                            local distanceOffset = (Vector2.new(screenCoords.X, screenCoords.Y) - midpoint).Magnitude
                            if distanceOffset < minimumDistance then
                                minimumDistance = distanceOffset
                                chosenTargetName = entityName
                            end
                        end

                end
            end
        end
        return chosenTargetName

        for _, conn in ipairs(activeConnections) do

    local function updateTrees()
        local assets = workspace:FindFirstChild("world_assets")
        local Trees = assets 
            and assets:FindFirstChild("StaticObjects") 
            and assets.StaticObjects:FindFirstChild("Trees")

        if not Trees then return end

        for _, t in ipairs(Trees:GetChildren()) do
            if t:IsA("Model") and not t.Name:match("^Bush") then
                for _, part in ipairs(t:GetDescendants()) do
                    if part:IsA("MeshPart") then
                        local n = part.Name
                        if n:find("Leaves") then
                            part.Transparency = CustomWorldVisualsEnabled and LeavesTransparency or 0
                        elseif n:find("Trunk") then
                            part.Transparency = CustomWorldVisualsEnabled and TrunkTransparency or 0
                        end
                    end
                end

        if workspace:FindFirstChild("Terrain") then

                sethiddenproperty(workspace.Terrain, "Decoration", not (CustomWorldVisualsEnabled and GrassTransparency >= 1))

        end

        updateTrees()
        updateGrass()
    end

        TracerThickness = 0.2,
        TracerTransparency = 0,
        TracerColor = Color3.fromRGB(0, 150, 255)
    }

    local CharacterState = require(game:GetService("ReplicatedStorage"):WaitForChild("CustomCharacter"):WaitForChild("CustomCharacterState"))
    local Entities = workspace:WaitForChild("game_assets", 10):WaitForChild("Entities", 10)

    local function getWeaponPosition()
        local cam = workspace.CurrentCamera
        local fallbackPos = cam and (cam.CFrame.Position + (cam.CFrame.LookVector * 8)) or Vector3.zero
        
        if not Entities then return fallbackPos end
        
        local bestEntity = nil
        local shortestDist = math.huge
        
        for _, v in ipairs(Entities:GetChildren()) do
            if v:IsA("Model") and v:FindFirstChild("HumanoidRootPart") and v:FindFirstChild("RightHand") then
                if cam then
                    local dist = (v.HumanoidRootPart.Position - cam.CFrame.Position).Magnitude
                    if dist < shortestDist and dist < 12 then 
                        shortestDist = dist
                        bestEntity = v
                    end
                end
            end
        end

        local basePos = fallbackPos
        if bestEntity then
            local rightHand = bestEntity:FindFirstChild("RightHand")
            if rightHand then
                local rightHandBone = rightHand:FindFirstChild("RightHandBone")
                basePos = rightHandBone and rightHandBone.WorldPosition or rightHand.Position
            elseif bestEntity:FindFirstChild("Head") then
                basePos = bestEntity.Head.Position
            end
        end

        -- Push the start point 8 studs forward to completely avoid any mesh/camera clipping artifacts
        if cam then
            return basePos + (cam.CFrame.LookVector * 8)
        end
        
        return basePos
    end

    local function createTracer(targetHead)
        if not Config.TracerEnabled or not targetHead then return end
        
        local startWorldPos = getWeaponPosition()
        local endWorldPos = targetHead.Position

        local tracerPart = Instance.new("Part")
        tracerPart.Size = Vector3.new(0.1, 0.1, 0.1)
        tracerPart.Transparency = 1
        tracerPart.Anchored = true
        tracerPart.CanCollide = false
        tracerPart.CFrame = CFrame.new(startWorldPos)
        tracerPart.Parent = workspace

        local att1 = Instance.new("Attachment", tracerPart)
        att1.Position = Vector3.zero

        local endPart = Instance.new("Part")
        endPart.Size = Vector3.new(0.1, 0.1, 0.1)
        endPart.Transparency = 1
        endPart.Anchored = true
        endPart.CanCollide = false
        endPart.CFrame = CFrame.new(endWorldPos)
        endPart.Parent = workspace

        local att2 = Instance.new("Attachment", endPart)
        att2.Position = Vector3.zero

        local beam = Instance.new("Beam")
        beam.Attachment0 = att1
        beam.Attachment1 = att2
        beam.Color = ColorSequence.new(Config.TracerColor)
        beam.Width0 = Config.TracerThickness
        beam.Width1 = Config.TracerThickness
        beam.FaceCamera = true
        beam.Transparency = NumberSequence.new(1)
        beam.Parent = tracerPart

        local elapsed = 0
        local lifetime = math.max(Config.TracerLifetime, 0.05)
        local maxAlpha = math.clamp(Config.TracerTransparency, 0, 1)

        local connection
        connection = RunService.RenderStepped:Connect(function(dt)
            elapsed = elapsed + dt
            local progress = elapsed / lifetime

            if progress >= 1 then
                connection:Disconnect()
                beam:Destroy()
                att1:Destroy()
                att2:Destroy()
                tracerPart:Destroy()
                endPart:Destroy()
                return
            end

            local alpha
            if progress < 0.2 then
                alpha = (progress / 0.2) * maxAlpha
            else
                alpha = (1 - ((progress - 0.2) / 0.8)) * maxAlpha
            end

            beam.Transparency = NumberSequence.new(math.clamp(alpha, 0, 1))
            beam.Width0 = Config.TracerThickness
            beam.Width1 = Config.TracerThickness
        end)
    end

                if not Config.Enabled or not Entities then return old(self, p5, p6, p7) end

                local state = CharacterState:Get(p5)
                if state then
                    local orig, best, bestdist = state.GetEyeTransform, nil, math.huge
                    local cam = workspace.CurrentCamera
                    local originPos = cam and cam.CFrame.Position

                    if originPos then
                        for _, v in Entities:GetChildren() do
                            if v:FindFirstChild("UpperTorso") or not v:FindFirstChild("HumanoidRootPart") or v.HumanoidRootPart.CollisionGroup == "PropGroup" or not v:FindFirstChild("Head") then continue end
                            
                            local targetPos = v.HumanoidRootPart.Position
                            local dist = (targetPos - originPos).Magnitude
                            
                            if dist <= Config.Distance and dist < bestdist then
                                bestdist = dist
                                best = v
                            end
                        end
                    end

                    state.GetEyeTransform = function(inst)
                        local tf = orig(inst)
                        if best then
                            local headPos = best.Head.Position
                            return CFrame.lookAt(headPos - (headPos - tf.Position).Unit * 0.5, headPos)
                        end
                        return tf
                    end
                    
                    local res = { old(self, p5, p6, p7) }
                    
                    if best and best:FindFirstChild("Head") then
                        createTracer(best.Head)
                    end

                    state.GetEyeTransform = orig
                    return table.unpack(res)
                end

                return old(self, p5, p6, p7)
            end))

    end

    local cachedRayParams = RaycastParams.new()
    cachedRayParams.FilterType = Enum.RaycastFilterType.Exclude
    cachedRayParams.IgnoreWater = true
    local cachedIgnore = {}

    local function IsVisible(targetPart, OriginPos, extraIgnore)
        if not targetPart or not targetPart.Parent then return false, nil end
        local camera = workspace.CurrentCamera
        OriginPos = OriginPos or (camera and camera.CFrame.Position) or Vector3.new()
        
        table.clear(cachedIgnore)
        if LocalPlayer and LocalPlayer.Character then
            table.insert(cachedIgnore, LocalPlayer.Character)
        end
        if camera then table.insert(cachedIgnore, camera) end
        if typeof(extraIgnore) == 'table' then
            for _, v in ipairs(extraIgnore) do 
                if typeof(v) == 'Instance' then 
                    table.insert(cachedIgnore, v) 
                end 
            end
        elseif typeof(extraIgnore) == 'Instance' then 
            table.insert(cachedIgnore, extraIgnore) 
        end
        
        cachedRayParams.FilterDescendantsInstances = cachedIgnore
        
        local direction = targetPart.Position - OriginPos
        if direction.Magnitude == 0 then return true, nil end
        
        local result = workspace:Raycast(OriginPos, direction, cachedRayParams)
        if not result then return true, nil end
        if result.Instance and result.Instance:IsDescendantOf(targetPart.Parent) then return true, result end
        return false, result
    end

    local function GetClosestTarget(fovRadius, aimPartName)
        local camera = workspace.CurrentCamera
        if not camera then return nil, nil, nil end

        local list = get_entity_list()
        if not list then return nil, nil, nil end

        local mousePos = UserInputService:GetMouseLocation()
        local maxDist = fovRadius
        local targetPart, bestPlayer, bestCollider

        for _, v in pairs(list) do
            local player = v.Player
            if player and player ~= LocalPlayer then
                local character = v.Character
                local worldmodel = v.WorldModel
                local root = v.RootPart

                if character and worldmodel then
                    local part = nil
                    if aimPartName == "Head" then
                        for _, obj in ipairs(character:GetDescendants()) do
                            if obj:GetAttribute("StretchedHead") then
                                part = obj
                                break
                            end
                        end
                    end
                    if not part then
                        part = worldmodel:FindFirstChild(aimPartName)
                    end

                    if part then
                        local sp, onScreen = camera:WorldToViewportPoint(part.Position)
                        if onScreen then
                            local dist = (Vector2.new(sp.X, sp.Y) - mousePos).Magnitude
                            if dist < maxDist then
                                maxDist = dist
                                targetPart = part
                                bestPlayer = player
                                bestCollider = root
                            end
                        end
                    end
                end
            end
        end
        return targetPart, bestPlayer, bestCollider
    end

    local function executeScan()
            local Players = cloneref(game:GetService("Players"))
            local HttpService = cloneref(game:GetService("HttpService"))

                ["Mulepack"] = true,
                ["DuffleBag"] = true,
                ["BackpackTier4"] = true,
                ["BackpackTier3"] = true,
            }

            local Map = {}
            for Index, Val in pairs(RawMap) do
                Map[Index:lower()] = Val
            end

            local Attributes = {
                ["Pants"] = "EquipmentPants",
                ["Shirt"] = "EquipmentShirt",
                ["Hat"] = "EquipmentHat",
                ["Vest"] = "EquipmentVest",
                ["Mask"] = "EquipmentMask",
                ["Backpack"] = "EquipmentBackpack",
                ["Shoes"] = "EquipmentShoes",
            }

            local Found = {}
            local LocalPlayer = Players.LocalPlayer

            for _, Player in pairs(Players:GetPlayers()) do
                if Player == LocalPlayer then
                    continue
                end

                for Name, Attribute in pairs(Attributes) do
                    local Attr = Player:GetAttribute(Attribute)
                    Attr = Attr and HttpService:JSONDecode(Attr) or {}

                    if not Attr.ClassName then
                        continue
                    end

                    local Text = Attr.ClassName:gsub(".item", "")

                    if not Map[Text:lower()] then
                        continue
                    end

                    table.insert(Found, {
                        Player = Player,
                        Gun = Text
                    })

                    local notifText = Player.Name .. " has " .. Text .. "!"
                    if Library and type(Library.Notify) == "function" then
                        Library:Notify(notifText)
                    end
                end

                local GunInventory = Player:FindFirstChild("GunInventory")

                if not GunInventory then
                    continue
                end

                for _, Slot in pairs(GunInventory:GetChildren()) do
                    local Gun = Slot.Value

                    if not Gun or not Map[Gun.Name:lower()] then
                        continue
                    end

                    local Mag = Slot:FindFirstChild("BulletsInMagazine")
                    local Reserve = Slot:FindFirstChild("BulletsInReserve")

                    if not Mag or not Reserve then
                        continue
                    end

                    local TotalAmmo = Mag.Value + Reserve.Value

                    if Gun.Name:lower() ~= "barret50" and TotalAmmo <= 50 then
                        continue
                    end

                    local displayNameObj = Gun:FindFirstChild("DisplayName")
                    local displayName = displayNameObj and displayNameObj.Value or Gun.Name
                    local gunDesc = string.format("%s with %d ammo", displayName, TotalAmmo)

                    table.insert(Found, {
                        Player = Player,
                        Gun = gunDesc
                    })

                    local notifText = string.format("%s has %s (%d ammo)", Player.Name, Gun.Name, TotalAmmo)
                    if Library and type(Library.Notify) == "function" then
                        Library:Notify(notifText)
                    end
                end
            end

            print(string.format("Found %d things that match", #Found))

            for _, Data in pairs(Found) do
                print(string.format("Player %s has %s", Data.Player.Name, Data.Gun))
            end
            
            if Library and type(Library.Notify) == "function" then
                Library:Notify("Scan Complete! Found " .. #Found .. " items. Check F9.")
            end
        end

            executeScan()
        end)

    local function get_local_chassis()
        if carFly.cachedChassis and carFly.cachedChassis.Parent then
            local config = carFly.cachedModel and carFly.cachedModel:FindFirstChild("Configuration")
            local driver = config and config:FindFirstChild("Driver")
            if driver and driver:IsA("ObjectValue") and driver.Value == LocalPlayer then
                return carFly.cachedChassis, carFly.cachedModel
            end
        end

        for _, obj in pairs(workspace:GetChildren()) do
            local chassis = obj:FindFirstChild("Chassis")
            local config = obj:FindFirstChild("Configuration")
            if chassis and chassis:IsA("BasePart") and config then
                local driver = config:FindFirstChild("Driver")
                if driver and driver:IsA("ObjectValue") and driver.Value == LocalPlayer then
                    carFly.cachedChassis = chassis
                    carFly.cachedModel = obj
                    return chassis, obj
                end

        end
        return nil, nil
    end

    local function setupConstraints(chassis)
        if not chassis then return end
        
        -- Clean up old ones if they exist
        if carFly.attachment then carFly.attachment:Destroy() end
        if carFly.linearVelocity then carFly.linearVelocity:Destroy() end
        if carFly.alignOrientation then carFly.alignOrientation:Destroy() end
        if carFly.vectorForce then carFly.vectorForce:Destroy() end

        -- Create attachment at chassis center
        local att = Instance.new("Attachment")
        att.Name = "CarFlyAttachment"
        att.Parent = chassis
        carFly.attachment = att

        -- Linear velocity for smooth movement
        local lv = Instance.new("LinearVelocity")
        lv.Name = "CarFlyVelocity"
        lv.Attachment0 = att
        lv.MaxForce = math.huge
        lv.VectorVelocity = Vector3.zero
        lv.RelativeTo = Enum.ActuatorRelativeTo.World
        lv.Parent = chassis
        carFly.linearVelocity = lv

        local ao = Instance.new("AlignOrientation")
        ao.Name = "CarFlyOrientation"
        ao.Attachment0 = att
        ao.Mode = Enum.OrientationAlignmentMode.OneAttachment
        ao.MaxTorque = math.huge
        ao.Responsiveness = 200
        ao.Parent = chassis
        carFly.alignOrientation = ao

        local vf = Instance.new("VectorForce")
        vf.Name = "CarFlyGravityCanceller"
        vf.Attachment0 = att
        vf.RelativeTo = Enum.ActuatorRelativeTo.World
        vf.Force = Vector3.new(0, chassis:GetMass() * workspace.Gravity, 0)
        vf.Parent = chassis
        carFly.vectorForce = vf
    end

    local function removeConstraints()
        if carFly.attachment then carFly.attachment:Destroy(); carFly.attachment = nil end
        if carFly.linearVelocity then carFly.linearVelocity:Destroy(); carFly.linearVelocity = nil end
        if carFly.alignOrientation then carFly.alignOrientation:Destroy(); carFly.alignOrientation = nil end
        if carFly.vectorForce then carFly.vectorForce:Destroy(); carFly.vectorForce = nil end
    end

    local function startCarFly()
        if carFly.conn then return end
        carFly.conn = RunService.Heartbeat:Connect(function(dt)
            if not carFly.enabled then return end
            local chassis, model = get_local_chassis()
            if not chassis then return end
            
            if not carFly.attachment or carFly.attachment.Parent ~= chassis then
                setupConstraints(chassis)
            end

            if chassis.Anchored then chassis.Anchored = false end
            
            local cam = workspace.CurrentCamera
            if not cam then return end
            local uis = UserInputService or game:GetService("UserInputService")
            
            local lookVec = cam.CFrame.LookVector
            local moveDir = lookVec
            
            if uis and uis:IsKeyDown(Enum.KeyCode.A) then
                moveDir = (moveDir - cam.CFrame.RightVector * 0.8).Unit
            elseif uis and uis:IsKeyDown(Enum.KeyCode.D) then
                moveDir = (moveDir + cam.CFrame.RightVector * 0.8).Unit
            end

            if carFly.type == "Old / Blocky" then
                carFly.dir = moveDir
                if carFly.linearVelocity then
                    carFly.linearVelocity.VectorVelocity = carFly.dir * carFly.speed
                end
                if carFly.alignOrientation then
                    local targetCF = CFrame.lookAt(Vector3.zero, carFly.dir)
                    if carFly.backward then targetCF = targetCF * CFrame.Angles(0, math.pi, 0) end
                    carFly.alignOrientation.CFrame = targetCF
                end
            else
                carFly.dir = carFly.dir:Lerp(moveDir, math.clamp(dt * 10, 0, 1))
                if carFly.linearVelocity then
                    carFly.linearVelocity.VectorVelocity = carFly.linearVelocity.VectorVelocity:Lerp(carFly.dir * carFly.speed, math.clamp(dt * 12, 0, 1))
                end
                if carFly.alignOrientation then
                    local targetCF = CFrame.lookAt(Vector3.zero, carFly.dir)
                    if carFly.backward then targetCF = targetCF * CFrame.Angles(0, math.pi, 0) end
                    carFly.alignOrientation.CFrame = carFly.alignOrientation.CFrame:Lerp(targetCF, math.clamp(dt * 14, 0, 1))
                end
            end
        end)
    end

    local function stopCarFly()
        if carFly.conn then carFly.conn:Disconnect(); carFly.conn = nil end
        
        local chassis, _ = get_local_chassis()
        removeConstraints()
        
        if chassis and chassis.Parent then
            chassis.AssemblyLinearVelocity = Vector3.zero
            chassis.AssemblyAngularVelocity = Vector3.zero
        end

        carFly.cachedChassis = nil
        carFly.cachedModel = nil
        carFly.dir = Vector3.new(0, 0, 1)
    end

    local function IsVisible(targetPart, OriginPos, extraIgnore)
        if not targetPart or not targetPart.Parent then return false, nil end
        local camera = workspace.CurrentCamera
        OriginPos = OriginPos or (camera and camera.CFrame.Position) or Vector3.new()
        
        table.clear(cachedIgnore)
        if LocalPlayer and LocalPlayer.Character then
            table.insert(cachedIgnore, LocalPlayer.Character)

        if typeof(extraIgnore) == 'table' then
            for _, v in ipairs(extraIgnore) do

                    table.insert(cachedIgnore, v) 
                end 
            end
        elseif typeof(extraIgnore) == 'Instance' then 
            table.insert(cachedIgnore, extraIgnore) 
        end
        
        cachedRayParams.FilterDescendantsInstances = cachedIgnore
        
        local direction = targetPart.Position - OriginPos
        if direction.Magnitude == 0 then return true, nil end
        
        local result = workspace:Raycast(OriginPos, direction, cachedRayParams)
        if not result then return true, nil end
        if result.Instance and result.Instance:IsDescendantOf(targetPart.Parent) then return true, result end
        return false, result
    end

                            end
                        end
                    end
                end
            end
        end

    end

    local function isValidWeaponPart(descendant)
        if not descendant or not descendant:IsA("BasePart") then return false end
        
        local nameLower = descendant.Name:lower()
        
        if nameLower:find("hitbox") or nameLower:find("box") or nameLower:find("root") or
        nameLower:find("humano") or nameLower:find("collider") or nameLower:find("region") then
            return false
        end
        
        if descendant.Transparency >= 0.9 or descendant.Size.Magnitude < 0.05 then
            return false
        end
        
        return true
    end

    local function processGunDescendant(descendant)
        if not flags.GunChamsEnabled or not descendant then return end

            descendant.Material = flags.GunChamsMaterial

            descendant.Transparency = 0

                descendant.TextureID = ""

            descendant.MeshId = ""
            descendant.TextureId = ""

            local parent = descendant.Parent
            if parent and not parent:IsA("SurfaceGui") and not parent:IsA("BillboardGui") then

    -- Strictly target only the main 'Skin' part for arm chams and hide duplicate ghost limbs
    local function applyArmChams()
        local camera = Workspace.CurrentCamera or Workspace.Camera
        if not camera then return end
        
        local currentWeapon = camera:FindFirstChild('CurrentWeapon')
        if not currentWeapon then return end
        
        for _, v in ipairs(currentWeapon:GetDescendants()) do
            if v and v:IsA('BasePart') then
                local vName = v.Name:lower()
                
                if flags.ArmChamsEnabled then
                    if vName == 'skin' then
                        v.Material = Enum.Material.ForceField
                        v.Color = flags.ArmChamsColor
                        v.Transparency = 0
                    elseif vName:find('arm') or vName:find('hand') then
                        -- Hide duplicate wrapper limbs to prevent overlapping ghost arms
                        v.Transparency = 1
                    end
                    
                    if v.Name == 'Shirt' then
                        v:Destroy()
                    end
                else
                    if vName == 'skin' then
                        v.Material = Enum.Material.Plastic
                        v.Transparency = 0
                    elseif vName:find('arm') or vName:find('hand') then
                        v.Transparency = 0
                    end
                end
            end
        end
    end

    local bodyChamsCache = { surfaces = {}, parts = {} }

    local function processCharacterPart(descendant)
        if not descendant then return end
        local partName = descendant.Name:lower()
        if partName:find("weapon") or partName:find("gun") or partName:find("tool") or partName:find("hold") then 
            return 
        end

        if descendant:IsA("BasePart") then
            if flags.BodyChamsEnabled then
                if not bodyChamsCache.parts[descendant] then
                    bodyChamsCache.parts[descendant] = {
                        Material = descendant.Material,
                        Color = descendant.Color,
                        Transparency = descendant.Transparency,
                        TextureID = descendant:IsA("MeshPart") and descendant.TextureID or nil
                    }
                end
                
                if partName ~= "head" then
                    descendant.Material = flags.BodyChamsMaterial
                    descendant.Color = flags.BodyChamsColor
                    if descendant:IsA("MeshPart") then
                        descendant.TextureID = ""
                    end
                end
            else
                local stored = bodyChamsCache.parts[descendant]
                if stored then
                    descendant.Material = stored.Material
                    descendant.Color = stored.Color
                    descendant.Transparency = stored.Transparency
                    if descendant:IsA("MeshPart") and stored.TextureID then
                        descendant.TextureID = stored.TextureID
                    end
                    bodyChamsCache.parts[descendant] = nil
                end
            end
        elseif descendant:IsA("SurfaceAppearance") and flags.BodyChamsEnabled then
            if not bodyChamsCache.surfaces[descendant] then
                bodyChamsCache.surfaces[descendant] = { Parent = descendant.Parent }
            end
            descendant.Parent = nil
        elseif (descendant:IsA("Clothing") or descendant:IsA("Accessory")) and flags.BodyChamsEnabled then
            descendant.Transparency = 1
        end
    end

    local function applyToPlayerCharacter(entity)
        if not entity then return end

        if flags.BodyChamsEnabled then
            local head = entity:FindFirstChild("Head")
            if head and head:IsA("BasePart") then
                if not bodyChamsCache.parts[head] then
                    bodyChamsCache.parts[head] = {
                        Material = head.Material,
                        Color = head.Color,
                        Transparency = head.Transparency
                    }
                end
                head.Material = flags.BodyChamsMaterial
                head.Color = flags.BodyChamsColor
                head.Transparency = flags.BodyHeadTransparency
            end

            for _, descendant in pairs(entity:GetDescendants()) do
                processCharacterPart(descendant)
            end
        else
            for surfaceApp, data in pairs(bodyChamsCache.surfaces) do
                if data.Parent and data.Parent:IsDescendantOf(entity) then
                    surfaceApp.Parent = data.Parent
                    bodyChamsCache.surfaces[surfaceApp] = nil
                end
            end
            
            for part, data in pairs(bodyChamsCache.parts) do
                if part and part.Parent and part:IsDescendantOf(entity) then
                    part.Material = data.Material
                    part.Color = data.Color
                    if data.Transparency then part.Transparency = data.Transparency end
                    if part:IsA("MeshPart") and data.TextureID then part.TextureID = data.TextureID end
                    bodyChamsCache.parts[part] = nil
                end
            end
        end
    end

    local function applyVehicleChams(vehicle)
        if not vehicle or carChamsState.vehicles[vehicle] then return end
        
        local data = {parts = {}, originals = {}}
        for _, part in ipairs(vehicle:GetDescendants()) do
            if part and part:IsA("BasePart") and part.Transparency < 1 then
                data.originals[part] = {Material = part.Material, Color = part.Color, Transparency = part.Transparency}
                part.Material = flags.CarChamsMaterial
                part.Color = flags.CarChamsColor
                part.Transparency = flags.CarChamsTransparency
                
                local sa = part:FindFirstChildOfClass("SurfaceAppearance")
                if sa then pcall(function() sa:Destroy() end) end
                table.insert(data.parts, part)
            end
        end
        
        data.connection = vehicle.DescendantAdded:Connect(function(desc)
            if desc and desc:IsA("BasePart") then
                data.originals[desc] = {Material = desc.Material, Color = desc.Color, Transparency = desc.Transparency}
                desc.Material = flags.CarChamsMaterial
                desc.Color = flags.CarChamsColor
                desc.Transparency = flags.CarChamsTransparency
                local sa = desc:FindFirstChildOfClass("SurfaceAppearance")
                if sa then pcall(function() sa:Destroy() end) end
            end
        end)
        
        carChamsState.vehicles[vehicle] = data
    end

    local function removeVehicleChams(vehicle)
        local data = carChamsState.vehicles[vehicle]
        if not data then return end
        if data.connection then data.connection:Disconnect() end
        for part, orig in pairs(data.originals) do
            if part and part.Parent then
                part.Material = orig.Material
                part.Color = orig.Color
                part.Transparency = orig.Transparency
            end
        end
        carChamsState.vehicles[vehicle] = nil
    end

    local function clearVehicleCache()

    local function getMeshPartsInfo(model)
        local info = {}

        for _, child in ipairs(model:GetChildren()) do
            if child:IsA("MeshPart") then
                local childNames = {}

                for _, childObject in ipairs(child:GetChildren()) do
                    table.insert(childNames, childObject.Name)
                end

                table.sort(childNames)

                table.insert(info, {
                    Name = child.Name,
                    MeshId = child.MeshId,
                    ChildrenNames = childNames
                })
            end
        end

        table.sort(info, function(a, b)
            return a.Name < b.Name
        end)

        return info
    end

    local function meshPartsMatch(a, b)
        if #a ~= #b then
            return false
        end

        for i = 1, #a do
            if a[i].Name ~= b[i].Name
                or a[i].MeshId ~= b[i].MeshId then

                return false
            end

            local c1 = a[i].ChildrenNames
            local c2 = b[i].ChildrenNames

            if #c1 ~= #c2 then
                return false
            end

            for j = 1, #c1 do
                if c1[j] ~= c2[j] then
                    return false
                end
            end
        end

        return true
    end

    local function getChildrenNames(model)
        local names = {}

        for _, child in ipairs(model:GetChildren()) do
            table.insert(names, child.Name)
        end

        table.sort(names)

        return names
    end

    local function childrenNamesMatch(a, b)
        if #a ~= #b then
            return false
        end

        for i = 1, #a do
            if a[i] ~= b[i] then
                return false
            end
        end

        return true
    end

    local function getDescendantsNames(model)
        local names = {}

        for _, descendant in ipairs(model:GetDescendants()) do

    end

        end

        return true
    end

    local function checkModelAgainstGunData(model)
        if not model:IsA("Model") then
            return nil
        end

        local meshInfo = getMeshPartsInfo(model)

        if #meshInfo > 0 then
            for name, referenceInfo in pairs(gunDataMeshInfos) do
                if meshPartsMatch(meshInfo, referenceInfo) then
                    return name
                end
            end
        else
            local childNames = getChildrenNames(model)
            local candidates = {}

            for name in pairs(gunDataMeshInfos) do
                local gunData = gunDataFolder:FindFirstChild(name)

                if gunData then
                    local worldModel = gunData:FindFirstChild("WorldModel")

                    if worldModel then
                        local referenceChildren = getChildrenNames(worldModel)

                        if childrenNamesMatch(
                            childNames,
                            referenceChildren
                        ) then
                            table.insert(candidates, name)
                        end
                    end
                end
            end

            if #candidates == 1 then
                return candidates[1]
            elseif #candidates > 1 then
                local descendantNames = getDescendantsNames(model)

                for _, name in ipairs(candidates) do
                    local gunData = gunDataFolder:FindFirstChild(name)

                    if gunData then
                        local worldModel = gunData:FindFirstChild("WorldModel")

                        if worldModel then
                            if descendantsNamesMatch(
                                descendantNames,
                                getDescendantsNames(worldModel)
                            ) then
                                return name
                            end
                        end
                    end
                end
            end
        end

        return nil
    end

        if flags.GunESPEnabled then
            for _, model in ipairs(miscFolder:GetChildren()) do
                scanGun(model)
            end
        end

        if flags.RepairKitEnabled then
            for _, object in ipairs(miscFolder:GetDescendants()) do
                scanRepairKit(object)
            end
        end
    end

    local function hideCharacters(list)
        for _, object in pairs(list or {}) do
            object.Visible = false
        end
    end

    local function drawAnimatedText(
        text,
        chars,
        position,
        size,
        color1,
        color2
    )
        local charWidth = size * 0.55
        local totalWidth = #text * charWidth
        local startX = position.X - totalWidth / 2

        for i = 1, #text do
            local charObject = chars[i]

            if not charObject then
                charObject = Drawing.new("Text")
                charObject.Center = false
                charObject.Outline = true
                charObject.OutlineColor = Color3.fromRGB(15, 15, 15)
                charObject.Font = 2

                chars[i] = charObject
            end

            charObject.Text = text:sub(i, i)
            charObject.Size = size
            charObject.Transparency = 1
            charObject.Color = GetWaveColor(
                color1,
                color2,
                i,
                5,
                0.4
            )

            charObject.Position = Vector2.new(
                startX + ((i - 1) * charWidth),
                position.Y
            )

            charObject.Visible = true
        end

        for i = #text + 1, #chars do
            chars[i].Visible = false
        end
    end

    local function hideGunData(data)
        data.billboard.Enabled = false
        data.nameLabel.Visible = false
        data.distanceLabel.Visible = false

        hideCharacters(data.nameChars)
        hideCharacters(data.distanceChars)
    end

    local function hideRepairData(data)
        data.billboard.Enabled = false
        data.label.Visible = false
        data.distanceLabel.Visible = false

        hideCharacters(data.nameChars)
        hideCharacters(data.distanceChars)
    end

    local function updateGunESP()
        Camera = workspace.CurrentCamera or Camera

        if not Camera then
            return
        end

        local character = LocalPlayer.Character
        local root = character and character:FindFirstChild("HumanoidRootPart")

        local origin = root
            and root.Position
            or Camera.CFrame.Position

        for model, data in pairs(weaponESP.Active) do
            if not model.Parent
                or not data.handle
                or not data.handle.Parent then

                if data.billboard then
                    pcall(function()
                        data.billboard:Destroy()
                    end)
                end

                hideCharacters(data.nameChars)
                hideCharacters(data.distanceChars)

                weaponESP.Active[model] = nil
                continue
            end

            local distance = (data.handle.Position - origin).Magnitude

            if not flags.GunESPEnabled
                or distance > flags.GunRenderDistance then

                hideGunData(data)
                continue
            end

            local viewport, onScreen = Camera:WorldToViewportPoint(
                data.handle.Position + Vector3.new(0, 2.5, 0)
            )

            if not onScreen then
                hideGunData(data)
                continue
            end

            local yOffset = 0
            local hasContent = false

            if flags.GunNameEnabled then
                local text = "[" .. data.name .. "]"

                if flags.GunNameAnimated then
                    data.nameLabel.Visible = false

                    drawAnimatedText(
                        text,
                        data.nameChars,
                        Vector2.new(
                            viewport.X,
                            viewport.Y + yOffset
                        ),
                        14,
                        flags.GunNameColor,
                        flags.GunNameColor2
                    )
                else
                    hideCharacters(data.nameChars)

                    data.nameLabel.Text = text
                    data.nameLabel.TextColor3 = flags.GunNameColor
                    data.nameLabel.TextSize = 14
                    data.nameLabel.Position = UDim2.new(
                        0,
                        viewport.X,
                        0,
                        viewport.Y + yOffset
                    )
                    data.nameLabel.Visible = true
                end

                yOffset = yOffset + 15
                hasContent = true
            else
                data.nameLabel.Visible = false
                hideCharacters(data.nameChars)
            end

            if flags.GunDistanceEnabled then
                local distanceText = string.format(
                    "[%d M]",
                    math.floor(distance / 3.5461)
                )

                if flags.GunDistanceAnimated then
                    data.distanceLabel.Visible = false

                    drawAnimatedText(
                        distanceText,
                        data.distanceChars,
                        Vector2.new(
                            viewport.X,
                            viewport.Y + yOffset
                        ),
                        14,
                        flags.GunDistanceColor,
                        flags.GunDistanceColor2
                    )
                else
                    hideCharacters(data.distanceChars)

                    data.distanceLabel.Text = distanceText
                    data.distanceLabel.TextColor3 = flags.GunDistanceColor
                    data.distanceLabel.TextSize = 14
                    data.distanceLabel.Position = UDim2.new(
                        0,
                        viewport.X,
                        0,
                        viewport.Y + yOffset
                    )
                    data.distanceLabel.Visible = true
                end

                hasContent = true
            else
                data.distanceLabel.Visible = false
                hideCharacters(data.distanceChars)
            end

            data.billboard.Enabled = hasContent
        end

        for part, data in pairs(repairKitESP.Active) do
            if not part.Parent then
                if data.billboard then
                    pcall(function()
                        data.billboard:Destroy()
                    end)
                end

                hideCharacters(data.nameChars)
                hideCharacters(data.distanceChars)

                repairKitESP.Active[part] = nil
                continue
            end

            local distance = (part.Position - origin).Magnitude

            if not flags.RepairKitEnabled
                or distance > flags.GunRenderDistance then

                hideRepairData(data)
                continue
            end

            local viewport, onScreen = Camera:WorldToViewportPoint(
                part.Position + Vector3.new(0, 2.5, 0)
            )

            if not onScreen then
                hideRepairData(data)
                continue
            end

            local yOffset = 0
            local hasContent = false

            if flags.RepairKitEnabled then
                local text = "[Repair Kit]"

                if flags.RepairKitAnimated then
                    data.label.Visible = false

                    drawAnimatedText(
                        text,
                        data.nameChars,
                        Vector2.new(
                            viewport.X,
                            viewport.Y + yOffset
                        ),
                        14,
                        flags.RepairKitColor,
                        flags.RepairKitColor2
                    )
                else
                    hideCharacters(data.nameChars)

                    data.label.Text = text
                    data.label.TextColor3 = flags.RepairKitColor
                    data.label.TextSize = 14
                    data.label.Position = UDim2.new(
                        0,
                        viewport.X,
                        0,
                        viewport.Y + yOffset
                    )
                    data.label.Visible = true
                end

                yOffset = yOffset + 15
                hasContent = true
            else
                data.label.Visible = false
                hideCharacters(data.nameChars)
            end

            if flags.RepairKitDistanceEnabled then
                local distanceText = string.format(
                    "[%d M]",
                    math.floor(distance / 3.5461)
                )

                if flags.RepairKitDistanceAnimated then
                    data.distanceLabel.Visible = false

                    drawAnimatedText(
                        distanceText,
                        data.distanceChars,
                        Vector2.new(
                            viewport.X,
                            viewport.Y + yOffset
                        ),
                        14,
                        flags.RepairKitDistanceColor,
                        flags.RepairKitDistanceColor2
                    )
                else
                    hideCharacters(data.distanceChars)

                    data.distanceLabel.Text = distanceText
                    data.distanceLabel.TextColor3 = flags.RepairKitDistanceColor
                    data.distanceLabel.TextSize = 14
                    data.distanceLabel.Position = UDim2.new(
                        0,
                        viewport.X,
                        0,
                        viewport.Y + yOffset
                    )
                    data.distanceLabel.Visible = true
                end

                hasContent = true
            else
                data.distanceLabel.Visible = false
                hideCharacters(data.distanceChars)
            end

            data.billboard.Enabled = hasContent
        end
    end

    -- Scan existing objects once.
    task.spawn(function()
        scan()
    end)

    -- New objects are handled individually instead of rescanning the
    -- entire Misc folder.
    miscFolder.ChildAdded:Connect(function(object)
        task.delay(0.1, function()
            if not object.Parent then
                return
            end

            if flags.GunESPEnabled then
                scanGun(object)
            end

            if flags.RepairKitEnabled then
                scanRepairKit(object)

                for _, descendant in ipairs(object:GetDescendants()) do
                    scanRepairKit(descendant)
                end
            end
        end)
    end)

    miscFolder.DescendantAdded:Connect(function(object)
        if not flags.RepairKitEnabled then
            return
        end

        task.defer(function()
            if object.Parent then
                scanRepairKit(object)
            end
        end)
    end)

    -- Only updates already-found ESP objects.
    -- No expensive world scan happens here.
    RunService.RenderStepped:Connect(function()
        updateGunESP()
    end)

do
    local RunService = game:GetService("RunService")
    Library:SetWatermarkVisibility(true)
    local FrameTimer = tick()
    local FrameCounter = 0
    local FPS = 60
    local StartTime = tick()

    local executorName = "Unknown"
    pcall(function()
        if identifyexecutor then
            executorName = identifyexecutor()
        elseif getexecutorname then
            executorName = getexecutorname()
        end
    end)

    RunService.RenderStepped:Connect(function()
        FrameCounter += 1
        if (tick() - FrameTimer) >= 1 then
            FPS = FrameCounter
            FrameTimer = tick()
            FrameCounter = 0
        end

        local ping = 0
        pcall(function()
            ping = math.floor(game:GetService('Stats').Network.ServerStatsItem['Data Ping']:GetValue())
        end)

        local playerCount = #game:GetService("Players"):GetPlayers()
        local elapsedSeconds = math.floor(tick() - StartTime)
        local minutes = math.floor(elapsedSeconds / 60)
        local seconds = elapsedSeconds % 60
        local timerFormatted = string.format("%02d:%02d", minutes, seconds)

        Library:SetWatermark(('Puppyware | %s | %s fps | %s ms | Players: %s | Time: %s'):format(
            executorName,
            math.floor(FPS),
            ping,
            playerCount,
            timerFormatted
        ))
    end)
end
