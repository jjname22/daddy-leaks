        local function changeMetatable(fn)
            local wasReadonly
            if type(env.isreadonly) == "function" then
                local ok, value = pcall(env.isreadonly, mt)
                if not ok or type(value) ~= "boolean" then return false end
                wasReadonly = value
                if protectAfterChanges == nil then protectAfterChanges = value end
            end
            -- Do not unlock a metatable when its original protection cannot be restored.
            if wasReadonly == true then
                if type(env.setreadonly) ~= "function" then return false end
                if not pcall(env.setreadonly, mt, false) then return false end
            end
            local ok = pcall(fn)
            if protectAfterChanges == true then
                local locked = pcall(env.setreadonly, mt, true)
                ok = ok and locked
            end
            return ok
        end

        local function restoreHeartbeat()
            if deactivateHeartbeat then deactivateHeartbeat(); deactivateHeartbeat = nil end
            if not mt then return end

                local ownsIndex = read(mt, "__index") == wrappedIndex
                local ownsNewIndex = read(mt, "__newindex") == wrappedNewIndex

                end

        end

                env.write(mt, "__newindex", wrappedNewIndex)

                control.heartbeatInstalled = true

                restoreHeartbeat()
                pcall(env.write, manager, "s", rawHeartbeat)
            end
        end

        local function disableErrorListeners()

                    -- Only take ownership of listeners whose previous state is known.

                        connection:Disable()

                    end

-- the live sequence counter and forces tamper reads to clean values, so the
-- v_b flush check can never trip even if Chroma's rawget hook is paused.

        local shared = env.shared

        }

        local PRIME = 16777619
        local MASK32 = 4294967295
        local MASK16 = 65535

        local function sign(num)
            num = tostring(num)
            local hash = 2166136261
            for i = 1, #num do
                hash = bit32.band(bit32.bxor(hash, string.byte(num, i)) * PRIME, MASK32)
            end
            return "\n>" .. num .. "--" .. bit32.band(hash, MASK16)
        end

        -- Re-derive the heartbeat whenever the AC's sequence counter advances.
        local function refreshHeartbeat()
            local seq = rawget(env.target, "c")
            if type(seq) ~= "number" then seq = 0 end
            if seq == control.lastSeq and control.heartbeat then
                return control.heartbeat
            end
            control.lastSeq = seq
            control.heartbeat = sign(seq)
            return control.heartbeat
        end

        function control:Start()
            if self.active then return end
            self.active = true

            if not self.installed then
                local mt = env.metatable(env.target)
                if type(mt) ~= "table" then
                    self.active = false
                    return
                end

                self.mt = mt
                self.oldIndex = rawget(mt, "__index")

                refreshHeartbeat()

                local wasReadonly = false
                if type(env.isreadonly) == "function" then
                    local ok, ro = pcall(env.isreadonly, mt)
                    if ok and ro == true then
                        wasReadonly = true
                        if type(env.setreadonly) == "function" then
                            pcall(env.setreadonly, mt, false)
                        end
                    end
                end

                rawset(mt, "__newindex", env.newcclosure(function(self, index, value)
                    if self == env.target then
                        if index == "s" then
                            refreshHeartbeat()
                            return
                        end
                        if index == "c" then
                            rawset(self, index, value)
                            control.lastSeq = value
                            control.heartbeat = sign(value)
                            return
                        end
                    end
                    rawset(self, index, value)
                end))

                rawset(mt, "__index", env.newcclosure(function(self, index)
                    if self == env.target then
                        if index == "s"  then return refreshHeartbeat() end
                        if index == "v"  then return "" end
                        if index == "vx" then return 0  end
                    end
                    local oi = self.oldIndex
                    if type(oi) == "function" then
                        return oi(self, index)
                    elseif type(oi) == "table" then
                        return rawget(oi, index)
                    end
                end))

                rawset(env.target, "s",  nil)
                rawset(env.target, "v",  "")
                rawset(env.target, "vx", 0)

                if wasReadonly and type(env.setreadonly) == "function" then
                    pcall(env.setreadonly, mt, true)
                end

                self.installed = true
            end

            self.resetThread = task.spawn(function()
                while control.active do
                    rawset(env.target, "vx", 0)
                    rawset(env.target, "v",  "")
                    task.wait(0)
                end
            end)

            self.refreshThread = task.spawn(function()
                while control.active do
                    refreshHeartbeat()
                    task.wait(0.5)
                end
            end)
        end

        function control:Destroy()
            if not self.active then return end
            self.active = false
            if self.resetThread then
                pcall(task.cancel, self.resetThread)
                self.resetThread = nil
            end
            if self.refreshThread then
                pcall(task.cancel, self.refreshThread)
                self.refreshThread = nil
            end
            -- Hooks intentionally left installed; unwinding them mid-session
            -- re-enables tamper accumulation and forces a v_b kick.
        end

        return control
    end

            shared      = getgenv(),

            metatable   = getrawmetatable,
            isreadonly  = isreadonly,
            setreadonly = setreadonly,

        })

        ServiceManagerBypassGeneration = ServiceManagerBypass.version
    end

		function MaidService.new()
			return setmetatable({
				_tasks = {},
				_features = {},
				_indices = {},
				_threads = {},
				_flagTasks = {}
			}, MaidService)
		end
		function MaidService:AddTask(task, feature)
			assert(task ~= nil, "Task cannot be nil")
			
			if feature then
				assert(type(feature) == "string" or type(feature) == "number", "Feature must be a string or number")
				self._features[feature] = self._features[feature] or {}
				table.insert(self._features[feature], task)
			end
			
			table.insert(self._tasks, task)
			
			if typeof(task) == "thread" then
				table.insert(self._threads, task)
			end
			
			return task
		end
		function MaidService:Add(task)
			assert(task ~= nil, "Task cannot be nil")
			return self:AddTask(task)
		end
		function MaidService:GiveTask(task)
			assert(task ~= nil, "Task cannot be nil")
			return self:AddTask(task)
		end
		function MaidService:GivePromise(promise)
			assert(promise ~= nil, "Promise cannot be nil")
			
			if promise.Status == "Rejected" or promise.Status == "Resolved" then
				return promise
			end
			
			local connection
			connection = promise:Finally(function()
				self:Remove(connection)
			end)
			
			return self:AddTask(connection)
		end
		function MaidService:Watch(flagRef, flagKey, task, feature) -- VIBECODED
			self:AddTask(task, feature)
			
			local flagTag = flagRef .. "." .. flagKey
			self._flagTasks = self._flagTasks or {}
			self._flagTasks[flagTag] = self._flagTasks[flagTag] or {}
			table.insert(self._flagTasks[flagTag], task)
			
			return task
		end
		function MaidService:CleanupFlag(flagRef, flagKey) -- VIBECODED
			if not self._flagTasks then return end
			local flagTag = flagRef .. "." .. flagKey
			local tasks = self._flagTasks[flagTag]
			if not tasks then return end
			for _, task in ipairs(tasks) do
				self:_cleanupTask(task)
				self:Remove(task)
			end
			self._flagTasks[flagTag] = {}
		end
		function MaidService:SetFlag(flags, key, value, onEnable, onDisable)
			flags[key] = value
			if not self._flagTasks then self._flagTasks = {} end  -- guard
			local flagTag = tostring(flags) .. "." .. key
			local tasks = self._flagTasks[flagTag] or {}
			for _, t in ipairs(tasks) do
				self:_cleanupTask(t)
				self:Remove(t)
			end
			self._flagTasks[flagTag] = {}
			if value and onEnable then
				local result = onEnable()
				if result then
					table.insert(self._flagTasks[flagTag], result)
					self:AddTask(result)
				end
			elseif not value and onDisable then
				onDisable()
			end
		end

			if task == nil then
				self:Remove(self._indices[index])
				self._indices[index] = nil
				return
			end
			
			self._indices[index] = task
			self:AddTask(task)
		end
		function MaidService:Remove(task)
			if task == nil then
				return
			end
			
			for i, v in ipairs(self._tasks) do
				if v == task then
					local taskToClean = table.remove(self._tasks, i)
					self:_cleanupTask(taskToClean)
					return
				end
			end
			
			for feature, tasks in pairs(self._features) do
				for i, v in ipairs(tasks) do
					if v == task then
						local taskToClean = table.remove(tasks, i)
						self:_cleanupTask(taskToClean)
						return
					end
				end
			end

				elseif task.Disconnect then
					task:Disconnect()
				elseif task.destroy then
					task:destroy()
				elseif task.disconnect then
					task:disconnect()
				elseif task.Clean then
					task:Clean()

					task:cancel()

--#region Writer
local Writer = {}
Writer.__index = Writer
function Writer.new(ForcedSize)
	return setmetatable({
		buf = buffer.create(ForcedSize or 128),
		offset = 0,
		capacity = ForcedSize or 128,
		forcedSize = ForcedSize,
	}, Writer)
end
function Writer:Ensure(size: number)
	local required = self.offset + size
	if required <= self.capacity then
		return
	end
	local NewSize = self.capacity
	while NewSize < required do
		NewSize += 32
	end
	local newBuffer = buffer.create(NewSize)
	buffer.copy(newBuffer, 0, self.buf, 0, self.offset)
	self.buf = newBuffer
	self.capacity = NewSize
end
function Writer:Build(): buffer
	local out = buffer.create(self.forcedSize or self.offset)
	buffer.copy(out, 0, self.buf, 0, self.offset)
	return out
end
function Writer:writeu8(v: number)
	self:Ensure(1)
	buffer.writeu8(self.buf, self.offset, v)
	self.offset += 1
end
function Writer:writeu16(v: number)
	self:Ensure(2)
	buffer.writeu16(self.buf, self.offset, v)
	self.offset += 2
end
function Writer:writeu32(v: number)
	self:Ensure(4)
	buffer.writeu32(self.buf, self.offset, v)
	self.offset += 4
end
function Writer:writei32(v: number)
	self:Ensure(4)
	buffer.writei32(self.buf, self.offset, v)
	self.offset += 4
end
function Writer:writef32(v: number)
	self:Ensure(4)
	buffer.writef32(self.buf, self.offset, v)
	self.offset += 4
end
function Writer:writef64(v: number)
	self:Ensure(8)
	buffer.writef64(self.buf, self.offset, v)
	self.offset += 8
end
function Writer:writestring(v: string)
	local len = #v
	self:Ensure(2 + len)
	buffer.writeu16(self.buf, self.offset, len)
	self.offset += 2
	buffer.writestring(self.buf, self.offset, v)
	self.offset += len
end
function Writer:writeVector3(v: Vector3)
	self:Ensure(12)
	buffer.writef32(self.buf, self.offset, v.X)
	buffer.writef32(self.buf, self.offset + 4, v.Y)
	buffer.writef32(self.buf, self.offset + 8, v.Z)
	self.offset += 12
end

GetClosestPlayer = function()
	local MouseLocation = UserInputService:GetMouseLocation()
	local ClosestPlayer, Distance = nil, math.huge
	local is360 = FLAGS.aimbot.targeting_mode == "360"
	local isRage = FLAGS.aimbot.mode == "Rage"

	for _, Data in CustomMeshCharacter:GetCharacters() do
		local Player = Data.Player
		if Player == LocalPlayer or IsDead(Player) or IsInSquad(Player) then
			continue
		end
		local Char = Data.WorldModel
		local AimPart = Char:FindFirstChild(TargetPart)
		if not AimPart then
			continue
		end

		if is360 or isRage then
			local Dist = (AimPart.Position - CurrentCamera.CFrame.Position).Magnitude
			if Dist < Distance then
				Distance = Dist
				ClosestPlayer = {
					Player = Data.Player,
					WorldModel = Data.WorldModel,
					TargetPart = AimPart,
					Character = Data.Character,
				}
			end
		else
			local ScreenPos, OnScreen = CurrentCamera:WorldToViewportPoint(AimPart.Position)
			if not OnScreen then
				continue
			end
			local Dist = (Vector2.new(ScreenPos.X, ScreenPos.Y) - MouseLocation).Magnitude
			local vp = CurrentCamera.ViewportSize

			if FLAGS.aimbot.fov_circle then
				local deg = math.clamp(FLAGS.aimbot.fov_size or 90, 0.5, 179.5)
				local r = math.tan(math.rad(deg / 2)) / math.tan(math.rad(CurrentCamera.FieldOfView / 2)) * (vp.Y / 2)
				if Dist > r then
					continue
				end
			end

			if Distance < Dist then
				continue
			end
			Distance = Dist
			ClosestPlayer = {
				Player = Data.Player,
				WorldModel = Data.WorldModel,
				TargetPart = AimPart,
				Character = Data.Character,
			}
		end
	end
	return ClosestPlayer
end

		for _, p in ipairs(child:GetDescendants()) do
			if p:IsA("BasePart") and p.Name == "Collider" then
				AimPart = p
				break
			end
		end
		-- fallback: any BasePart
		if not AimPart then
			AimPart = child:FindFirstChildWhichIsA("BasePart", true)
		end
		if not AimPart then continue end

		if is360 then
			local Dist = (AimPart.Position - CurrentCamera.CFrame.Position).Magnitude
			if Dist < Distance then
				Distance = Dist
				ClosestHeli = { TargetPart = AimPart, Model = child, IsHeli = true }
			end
		else
			local ScreenPos, OnScreen = CurrentCamera:WorldToViewportPoint(AimPart.Position)
			if not OnScreen then continue end
			local Dist = (Vector2.new(ScreenPos.X, ScreenPos.Y) - MouseLocation).Magnitude
			if FLAGS.aimbot.fov_circle then
				local vp = CurrentCamera.ViewportSize
				local deg = math.clamp(FLAGS.aimbot.fov_size or 90, 0.5, 179.5)
				local r = math.tan(math.rad(deg / 2)) / math.tan(math.rad(CurrentCamera.FieldOfView / 2)) * (vp.Y / 2)
				if Dist > r then continue end
			end
			if Dist < Distance then
				Distance = Dist
				ClosestHeli = { TargetPart = AimPart, Model = child, IsHeli = true }
			end
		end
	end
	return ClosestHeli
end

SampleVelocity = function(Target)
	local isInstant = FLAGS.aimbot.exploits.instant_bullet and (os.clock() - NextInstant) > 0
    local Player = Target.Player
    local Current = GetTargetVelocity(Target)

    if not VelocityHistory[Player] then
        VelocityHistory[Player] = {}
    end

    local Samples = VelocityHistory[Player]

    if #Samples > 0 then
        local Last = Samples[#Samples]
        local LastMag = Last.Magnitude
        local CurrentMag = Current.Magnitude

        if LastMag > 2 and CurrentMag > 2 then
            local Dot = Last.Unit:Dot(Current.Unit)
            if Dot < 0.7 then
                table.clear(Samples)
            end
        end
    end

    table.insert(Samples, Current)

    if #Samples > MaxSamples then
        table.remove(Samples, 1)
    end

    local Sum = Vector3.zero
    local Weight = 0

    for i = 1, #Samples do
        local W = i
        Sum += Samples[i] * W
        Weight += W
    end

    if isInstant then 
        return Vector3.zero 
    end

    return Sum / Weight
end
SolveIntercept = function(Origin, TargetPos, TargetVel, Speed)
	local P = TargetPos - Origin
	local V = TargetVel
	local S = Speed
	local A = S * S - V:Dot(V)
	local B = -2 * P:Dot(V)
	local C = -P:Dot(P)
	local T = nil
	if math.abs(A) < 0.0001 then
		if math.abs(B) < 0.0001 then
			return nil
		end
		local T0 = -C / B
		if T0 > 0 then
			T = T0
		else
			return nil
		end
	else
		local Disc = B * B - 4 * A * C
		if Disc < 0 then
			return nil
		end
		local SqrtDisc = math.sqrt(Disc)
		local T1 = (-B + SqrtDisc) / (2 * A)
		local T2 = (-B - SqrtDisc) / (2 * A)
		if T1 > 0 and T2 > 0 then
			T = math.min(T1, T2)
		elseif T1 > 0 then
			T = T1
		elseif T2 > 0 then
			T = T2
		else
			return nil
		end
	end
	return T
end

PredictPosition = function(Target)
    local Hrp = Target.Character and Target.Character:FindFirstChild("Part")
    local AimPart = Target.TargetPart

    if not Hrp then
        return AimPart.Position
    end

    local Origin = GetOrigin()

    local SmoothedVel = SampleVelocity(Target)
    local AimOffset = AimPart.Position - Hrp.Position
    local Latency = LocalPlayer:GetNetworkPing() * 0.5

    local Humanoid = Target.Character:FindFirstChildWhichIsA("Humanoid")
    local Airborne = false

    if Humanoid then
        Airborne = Humanoid.FloorMaterial == Enum.Material.Air
    end

    local T = SolveIntercept(Origin, Hrp.Position, SmoothedVel, BulletSpeed)

    if not T then
        return AimPart.Position
    end

    local LatencyCorrection = math.min(Latency, T)
    local LatencyPos = Hrp.Position + SmoothedVel * LatencyCorrection
    local T2 = SolveIntercept(Origin, LatencyPos, SmoothedVel, BulletSpeed)

    if not T2 then
        T2 = T
    end

    local Predicted = LatencyPos + SmoothedVel * T2

    if Airborne then
        Predicted = Vector3.new(
            Predicted.X,
            Predicted.Y - 0.5 * workspace.Gravity * T2 * T2,
            Predicted.Z
        )
    end

    return Predicted + AimOffset + Vector3.yAxis * (BulletGravity * T2 ^ 2)
end

	local MouseDelta = Vector2.zero
	local FreecamPosition = Vector3.zero
	local FreecamRotation = Vector2.zero
	local CurrentChassis = nil

		if FreecamEnabled then 
			return 
		end
		local lookVector = CurrentCamera.CFrame.LookVector
		local pitch = math.asin(lookVector.Y)
		local yaw = math.atan2(lookVector.X, lookVector.Z) - PI
		
		FreecamPosition = CurrentCamera.CFrame.Position
		FreecamRotation = Vector2.new(pitch, yaw)
		CurrentCamera.CameraType = Enum.CameraType.Scriptable
		Services.ContextActionService:BindActionAtPriority("FCMouse", function(actionName, inputState, inputObject)
			local delta = inputObject.Delta
			MouseDelta = Vector2.new(-delta.y, -delta.x)
			return Enum.ContextActionResult.Sink
		end, false, Enum.ContextActionPriority.High.Value, Enum.UserInputType.MouseMovement)
		Modules.DisableCameraService:Reserve("Freecam")
		hookfunction(setPlayerState, function(...)
			return
		end)
		--local Car = CarValue.Value
		--if Car then
		--    CurrentChassis = Car:FindFirstChild("Chassis")
		--end
		FreecamEnabled = true
	end)
	CleanFreecam = LPH_NO_VIRTUALIZE(function()
		if not FreecamEnabled then
			return 
		end
		Services.UserInputService.MouseBehavior = Enum.MouseBehavior.Default
		CurrentCamera.CameraType = Enum.CameraType.Custom
		Services.ContextActionService:UnbindAction("FCMouse")
		Modules.DisableCameraService:Release("Freecam")
		if isfunctionhooked(setPlayerState) then
			restorefunction(setPlayerState)
		end
		FreecamEnabled = false
	end)
	local FindGroundParams = RaycastParams.new()
	
	FindGroundParams.FilterType = Enum.RaycastFilterType.Exclude
	FindGroundParams.FilterDescendantsInstances = { CurrentCamera };
	
	OnPostSimulationFreecam = LPH_NO_VIRTUALIZE(function(DeltaTime)
		debugProfileBegin("OnPostSimulationFreecam")
		if not FLAGS.misc.freecam then
			if FreecamEnabled then
				CleanFreecam()
			end
			debugProfileEnd()
			return
		end
		if not FreecamEnabled then
			InitFreecam()
		end
		if Modules.FreeMouseService:IsMouseFree() or Services.UserInputService:GetFocusedTextBox() then
			Services.UserInputService.MouseBehavior = Enum.MouseBehavior.Default
			Services.UserInputService.MouseIconEnabled = true
			debugProfileEnd()
			return
		elseif Services.UserInputService.MouseBehavior ~= Enum.MouseBehavior.LockCenter then
			Services.UserInputService.MouseBehavior = Enum.MouseBehavior.LockCenter
			Services.UserInputService.MouseIconEnabled = false
		end
		-- Ugly code from the original freecam I just cant be asked to make better
		local W,A,S,D,LShift,Space = Services.UserInputService:IsKeyDown(Enum.KeyCode.W),
								Services.UserInputService:IsKeyDown(Enum.KeyCode.A),
								Services.UserInputService:IsKeyDown(Enum.KeyCode.S),
								Services.UserInputService:IsKeyDown(Enum.KeyCode.D),
								Services.UserInputService:IsKeyDown(Enum.KeyCode.LeftShift),
								Services.UserInputService:IsKeyDown(Enum.KeyCode.Space)
		local MovementVector = Vector3.zero
		
		if W then 
			MovementVector += Vector3.new(0, 0, 1) 
		end
		if A then 
			MovementVector -= Vector3.new(1, 0, 0) 
		end
		if S then 
			MovementVector -= Vector3.new(0, 0, 1) 
		end
		if D then 
			MovementVector += Vector3.new(1, 0, 0) 
		end
		if Space then 
			MovementVector += Vector3.new(0, 1, 0) 
		end
		if LShift then 
			MovementVector -= Vector3.new(0, 1, 0) 
		end
		local MoveLength = math.sqrt(MovementVector.X * MovementVector.X + MovementVector.Z * MovementVector.Z)
		
		if MoveLength > 0 then
			MovementVector = Vector3.new(
				MovementVector.X / MoveLength,
				MovementVector.Y,
				MovementVector.Z / MoveLength
			)
		end
		local CameraCFrame = CurrentCamera.CFrame
		local LookVector = CameraCFrame.LookVector
		local RightVector = CameraCFrame.RightVector
		local UpVector = CameraCFrame.UpVector
		local FinalX = (LookVector.X * MovementVector.Z + RightVector.X * MovementVector.X + UpVector.X * MovementVector.Y) * FLAGS.misc.freecamSpeed or 0
		local FinalY = (LookVector.Y * MovementVector.Z + RightVector.Y * MovementVector.X + UpVector.Y * MovementVector.Y) * FLAGS.misc.freecamSpeed or 0
		local FinalZ = (LookVector.Z * MovementVector.Z + RightVector.Z * MovementVector.X + UpVector.Z * MovementVector.Y) * FLAGS.misc.freecamSpeed or 0
		
		if FinalX ~= FinalX then 
			FinalX = 0 
		end
		if FinalY ~= FinalY then 
			FinalY = 0 
		end
		if FinalZ ~= FinalZ then 
			FinalZ = 0 
		end
		local MousePan = MouseDelta * MouseSpeed
		MouseDelta = Vector2.new()
		
		local ZoomFactor = math.sqrt(math.tan(DEG2RAD*(70/2))/math.tan(DEG2RAD*(CurrentCamera.FieldOfView/2)))
		
		FreecamRotation = FreecamRotation + MousePan * Vector2.new(0.75, 1) * 8 * (math.min(DeltaTime, 0.05) / ZoomFactor)
		
		local rad90 = DEG2RAD*90
		
		FreecamRotation = Vector2.new(math.clamp(FreecamRotation.X, -rad90, rad90), FreecamRotation.Y % PI2)
		FreecamPosition += Vector3.new(FinalX, FinalY, FinalZ)
		
		for x = -1, 1 do
			for z = -1, 1 do
				local GridOffset = Vector3.new(x * 512, 0, z * 512)
				local GridPosition = FreecamPosition + GridOffset
				local RaycastStart = Vector3.new(GridPosition.X, 1500, GridPosition.Z)
				local Result = workspace:Raycast(RaycastStart, Vector3.new(0, -2500, 0), FindGroundParams)
				local StreamPos

					StreamPos = Result.Position
				else
					StreamPos = Vector3.new(GridPosition.X, GridPosition.Y - 100, GridPosition.Z)
				end
				task.spawn(function()
					pcall(LocalPlayer.RequestStreamAroundAsync, LocalPlayer, StreamPos)
				end)
			end
		end
		
		local MovementPart = findRootPart()
		MovementPart.AssemblyLinearVelocity = Vector3.zero
		if CurrentChassis then
			CurrentChassis.AssemblyLinearVelocity = Vector3.zero
			CurrentChassis.AssemblyAngularVelocity = Vector3.zero
		end
		CurrentCamera.CFrame = CFrame.new(FreecamPosition) * CFrame.fromOrientation(FreecamRotation.x, FreecamRotation.y, 0)
	
		debugProfileEnd()
	end)

local CarFly

do
    -- Car Fly controller: donor constraints and steering, with owned vehicle/session cleanup.
    local function install(env, settings)
        local connection, destroyed, chassis, model, originalAnchored
        local attachment, velocity, orientation, gravityForce
        local direction = Vector3.new(0, 0, 1)
        local scanElapsed = .25
        local function owned(car, part)
            if not car or not car.Parent or not part or part.Parent ~= car or not part:IsA("BasePart") then return false end
            local config = car:FindFirstChild("Configuration")
            local driver = config and config:FindFirstChild("Driver")
            return driver and driver:IsA("ObjectValue") and driver.Value == env.player
        end
        local function clear(resetVelocity)
            for _, object in pairs({velocity, orientation, gravityForce, attachment}) do
                pcall(function() object:Destroy() end)
            end
            attachment, velocity, orientation, gravityForce = nil, nil, nil, nil
            if chassis and chassis.Parent then
                pcall(function()
                    if resetVelocity and owned(model, chassis) then
                        chassis.AssemblyLinearVelocity = Vector3.zero
                        chassis.AssemblyAngularVelocity = Vector3.zero
                    end
                    if originalAnchored and not chassis.Anchored then chassis.Anchored = true end
                end)
            end
            chassis, model, originalAnchored = nil, nil, nil
            direction = Vector3.new(0, 0, 1)
        end
        local function findChassis(dt)
            scanElapsed += math.max(dt, 0)
            local current = env.currentCar()
            local currentPart = current and current:FindFirstChild("Chassis")
            if owned(current, currentPart) then return currentPart, current end
            if owned(model, chassis) and model:FindFirstChild("Chassis") == chassis then return chassis, model end
            if scanElapsed < .25 then return end
            scanElapsed = 0
            for _, car in pairs(env.children()) do
                local part = car:FindFirstChild("Chassis")
                if owned(car, part) then return part, car end
            end
        end
        local function setup(part, car)
            clear(false)
            chassis, model, originalAnchored = part, car, part.Anchored
            attachment = env.new("Attachment")
            attachment.Name = "CarFlyAttachment"; attachment.Parent = part
            velocity = env.new("LinearVelocity")
            velocity.Name = "CarFlyVelocity"; velocity.Attachment0 = attachment
            velocity.MaxForce = math.huge; velocity.VectorVelocity = Vector3.zero
            velocity.RelativeTo = Enum.ActuatorRelativeTo.World; velocity.Parent = part
            orientation = env.new("AlignOrientation")
            orientation.Name = "CarFlyOrientation"; orientation.Attachment0 = attachment
            orientation.Mode = Enum.OrientationAlignmentMode.OneAttachment
            orientation.MaxTorque = math.huge; orientation.Responsiveness = 200; orientation.Parent = part
            gravityForce = env.new("VectorForce")
            gravityForce.Name = "CarFlyGravityCanceller"; gravityForce.Attachment0 = attachment
            gravityForce.RelativeTo = Enum.ActuatorRelativeTo.World
            gravityForce.ApplyAtCenterOfMass = true; gravityForce.Parent = part
        end
        local function step(dt)
            local part, car = findChassis(dt)
            if not part then clear(false); return end
            local camera = env.camera()
            if not camera then clear(false); return end
            if chassis ~= part or not attachment or attachment.Parent ~= part
                or not velocity or velocity.Parent ~= part or not orientation or orientation.Parent ~= part
                or not gravityForce or gravityForce.Parent ~= part then setup(part, car) end
            if part.Anchored then part.Anchored = false end
            gravityForce.Force = Vector3.new(0, part:GetMass() * env.gravity(), 0)
            local moveDir = camera.CFrame.LookVector
            if env.keyDown(Enum.KeyCode.A) then
                moveDir = (moveDir - camera.CFrame.RightVector * .8).Unit
            elseif env.keyDown(Enum.KeyCode.D) then
                moveDir = (moveDir + camera.CFrame.RightVector * .8).Unit
            end
            if settings.carflytype == "Old / Blocky" then
                direction = moveDir
                velocity.VectorVelocity = direction * settings.carflyspeed
                local target = CFrame.lookAt(Vector3.zero, direction)
                if settings.carflybackward then target *= CFrame.Angles(0, math.pi, 0) end
                orientation.CFrame = target
            else
                direction = direction:Lerp(moveDir, math.clamp(dt * 10, 0, 1))
                if direction.Magnitude < .001 then direction = moveDir end
                velocity.VectorVelocity = velocity.VectorVelocity:Lerp(direction * settings.carflyspeed, math.clamp(dt * 12, 0, 1))
                local target = CFrame.lookAt(Vector3.zero, direction)
                if settings.carflybackward then target *= CFrame.Angles(0, math.pi, 0) end
                orientation.CFrame = orientation.CFrame:Lerp(target, math.clamp(dt * 14, 0, 1))
            end
        end
        local control = {}
        function control:SetEnabled(value)
            if destroyed then return end
            settings.carfly = value == true
            if not settings.carfly then
                if connection then connection:Disconnect(); connection = nil end
                clear(true)
            elseif not connection then
                scanElapsed = .25
                connection = env.connect(function(dt)
                    local ok = pcall(step, dt)
                    if not ok then self:SetEnabled(false) end
                end)
            end
        end
        function control:Destroy()
            if destroyed then return end
            self:SetEnabled(false); destroyed = true
        end
        return control
    end
    -- Car Fly adapter:
    local shared = getgenv()
    local previous = shared.__ChromaCarFly
    if previous then pcall(function() previous:Destroy() end) end
    CarFly = install({player = LocalPlayer,
        currentCar = function()
            local car = LocalPlayer:FindFirstChild("Car")
            return car and car.Value
        end,
        children = function() return workspace:GetChildren() end,
        camera = function() return workspace.CurrentCamera end,
        gravity = function() return workspace.Gravity end,
        keyDown = function(key) return Services.UserInputService:IsKeyDown(key) end,
        new = function(kind) return Instance.new(kind) end,
        connect = function(fn) return Services.RunService.Heartbeat:Connect(fn) end}, FLAGS.misc)
    shared.__ChromaCarFly = CarFly
    Maid:AddTask(function()
        CarFly:Destroy()
        if shared.__ChromaCarFly == CarFly then shared.__ChromaCarFly = nil end
    end, "CarFly")
end

cfloop = nil;
carballs = nil;
impacts = {};
doCar = function(car)
    if cfloop then cfloop:Disconnect(); cfloop = nil end
    if impacts.impact and impacts.ogparent then
        pcall(function() impacts.impact.Parent = impacts.ogparent end)
    end
    impacts = {}
	if not car then 
		return 
	end;
	local impact = nil;
	for i, v in ipairs(car:GetDescendants()) do 
		if v.Name == 'Impact' then 
			impact = v;
			break 
		end;
	end;
	impacts = {
		car = car;
		impact = impact;
		ogparent = impact and impact.Parent;
	};
	if FLAGS.misc.no_impact and impact then
		impact.Parent = nil;
	end;
	if cfloop then 
		cfloop:Disconnect() 
		cfloop = nil;
	end;
	if FLAGS.misc.no_impact then
		cfloop = Services.RunService.Heartbeat:Connect(function()
			local currentcar = LocalPlayer.Car.Value
			if not currentcar then 
				if impacts.impact and impacts.ogparent then 
					pcall(function()
						impacts.impact.Parent = impacts.ogparent
					end)
				end
				impacts = {}
				if cfloop then 
					cfloop:Disconnect()
					cfloop = nil 
				end 
				return 
			end

		end)
	end
end
StopCar = function()
	if cfloop then 
		cfloop:Disconnect()
		cfloop = nil;
	end;
	if carballs then 
		carballs:Disconnect()
		carballs = nil;
	end;
	if impacts.impact and impacts.ogparent then 
		pcall(function() 
			impacts.impact.Parent = impacts.ogparent
		end)
	end;
	impacts = {}
end
FindCar = function()
	if carballs then
		carballs:Disconnect()
		carballs = nil
	end
	if LocalPlayer.Car.Value then

function findminoffset(target, origin)
    local maxdist = FLAGS.aimbot.exploits.manipulation.manip_dist
    local step = 0.25

    local aimpos= target.TargetPart.Position

    local centerdir = (aimpos - origin).Unit
    local centerresult = workspace:Raycast(origin, centerdir * FLAGS.aimbot.autofire.distance, autofireparams)

    if centerresult and centerresult.Instance then
        if centerresult.Instance:IsDescendantOf(target.Character) then
            return Vector3.zero, false, centerresult.Instance
        end
    end

    local dist = step
    while dist <= maxdist do
        for i, v in pairs(cheatinfo.manip_corners) do
            if v == Vector3.zero then
                continue
            end
            local newoffset = v * dist
            local neworigin = origin + newoffset
            local dir = (aimpos - neworigin).Unit
            local result = workspace:Raycast(neworigin, dir * FLAGS.aimbot.autofire.distance, autofireparams)

            if result and result.Instance then
                if result.Instance:IsDescendantOf(target.Character) then
                    return newoffset, i ~= 1, result.Instance
                end
            end
        end
        dist += step
    end

    return nil, false, nil
end

do
    local GunDataFolder = Services.ReplicatedStorage:WaitForChild("GunSystemAssets"):WaitForChild("GunData")
    local GunOffsets = {}

    for _, data in GunDataFolder:GetChildren() do
        local stats = data:FindFirstChild("Stats")
        local offset = stats and stats:FindFirstChild("Offset")
        if offset then
            GunOffsets[data.Name] = offset
        end
    end

    local appliedOffset = Vector3.zero
    local offsetEnabled = false

    local function addOffset(vec)
        for _, offsetVal in GunOffsets do
            offsetVal.Value += vec
        end
    end

    local function setOffsetEnabled(enabled)
        if enabled == offsetEnabled then
            return
        end
        offsetEnabled = enabled
        if enabled then
            addOffset(appliedOffset)
        else
            addOffset(-appliedOffset)
        end
    end

    local function updateAxis(axis, value)
        if offsetEnabled then
            addOffset(-appliedOffset)
        end
        if axis == "x" then
            appliedOffset = Vector3.new(value, appliedOffset.Y, appliedOffset.Z)
        elseif axis == "y" then
            appliedOffset = Vector3.new(appliedOffset.X, value, appliedOffset.Z)
        elseif axis == "z" then
            appliedOffset = Vector3.new(appliedOffset.X, appliedOffset.Y, value)
        end
        if offsetEnabled then
            addOffset(appliedOffset)
        end
    end

    ViewmodelOffsetSetEnabled = setOffsetEnabled
    ViewmodelOffsetUpdateAxis = updateAxis
end

RunService:BindToRenderStep("ESPAddQueue", 1, function()
	if #pendingAdds == 0 then
		return
	end
	local i = 1
	while i <= #pendingAdds do
		local entry = pendingAdds[i]
		local player, worldModel = entry[1], entry[2]
		if not worldModel.Parent then
			table.remove(pendingAdds, i)
		elseif worldModel:FindFirstChildWhichIsA("BasePart") then
			table.remove(pendingAdds, i)
			local old = espWorldModels[player]
			if old and old ~= worldModel then
				EspLib:remove(old)
			end
			espWorldModels[player] = worldModel
			EspLib:add(worldModel, player)
		else
			i = i + 1
		end
	end
end)
local addEspEntry = function(player, worldModel)
	table.insert(pendingAdds, {player, worldModel})
end
enablePlayerESP = function()
	for _, data in Modules.CustomMeshCharacter:GetCharacters() do
		if data.Player == LocalPlayer then
			continue
		end
		if espWorldModels[data.Player] then
			continue
		end
		addEspEntry(data.Player, data.WorldModel)
	end
	local added = Modules.CustomMeshCharacter.CharacterAdded:Connect(function(player, character, worldModel)
		if player == LocalPlayer then
			return
		end
		addEspEntry(player, worldModel)
	end)
	local removed = Modules.CustomMeshCharacter.CharacterRemoved:Connect(function(player)
		local oldWorldModel = espWorldModels[player]
		if not oldWorldModel then
			return
		end
		task.defer(function()
			if espWorldModels[player] == oldWorldModel then
				EspLib:remove(oldWorldModel)
				espWorldModels[player] = nil
			end
		end)
	end)
	espConnections = {added, removed}
end
disablePlayerESP = function()
	pendingAdds = {}
	if espConnections then
		for _, c in espConnections do
			c:Disconnect()
		end
		espConnections = nil
	end
	for player, worldModel in espWorldModels do
		EspLib:remove(worldModel)
	end
	table.clear(espWorldModels)
end


-- Skeleton overlay integrated with Chroma's existing CustomMeshCharacter ESP entries.
-- Uses the same world models registered in espWorldModels so it layers with the current player ESP.
local skeletonESP = {
    enabled = false,
    color = Color3.fromRGB(255, 255, 255),
    thickness = 1,
    lines = {},
}
local SkeletonBones = {
    {"Head", "UpperTorso"}, {"UpperTorso", "LowerTorso"},
    {"UpperTorso", "LeftUpperArm"}, {"LeftUpperArm", "LeftLowerArm"}, {"LeftLowerArm", "LeftHand"},
    {"UpperTorso", "RightUpperArm"}, {"RightUpperArm", "RightLowerArm"}, {"RightLowerArm", "RightHand"},
    {"LowerTorso", "LeftUpperLeg"}, {"LeftUpperLeg", "LeftLowerLeg"}, {"LeftLowerLeg", "LeftFoot"},
    {"LowerTorso", "RightUpperLeg"}, {"RightUpperLeg", "RightLowerLeg"}, {"RightLowerLeg", "RightFoot"},
    -- R6 fallback connections (lines for missing R15 parts simply stay hidden).
    {"Head", "Torso"}, {"Torso", "Left Arm"}, {"Torso", "Right Arm"},
    {"Torso", "Left Leg"}, {"Torso", "Right Leg"},
}
local function getSkeletonLines(player)
    local lines = skeletonESP.lines[player]
    if lines then return lines end
    lines = {}
    for i = 1, #SkeletonBones do
        local line = Drawing.new("Line")
        line.Visible = false
        line.Color = skeletonESP.color
        line.Thickness = skeletonESP.thickness
        line.Transparency = 1
        lines[i] = line
    end
    skeletonESP.lines[player] = lines
    return lines
end
local function hideSkeletonLines(lines)
    for _, line in ipairs(lines) do line.Visible = false end
end
local function removeSkeletonLines(player)
    local lines = skeletonESP.lines[player]
    if lines then
        for _, line in ipairs(lines) do pcall(function() line:Remove() end) end
        skeletonESP.lines[player] = nil
    end
end
RunService:BindToRenderStep("ChromaSkeletonESP", 2, function()
    local camera = workspace.CurrentCamera or CurrentCamera
    if not skeletonESP.enabled or not camera then
        for _, lines in pairs(skeletonESP.lines) do hideSkeletonLines(lines) end
        return
    end
    for player, worldModel in pairs(espWorldModels) do
        if player == LocalPlayer or not worldModel or not worldModel.Parent then
            removeSkeletonLines(player)
        else
            local lines = getSkeletonLines(player)
            hideSkeletonLines(lines)
            for i, bone in ipairs(SkeletonBones) do
                local partA = worldModel:FindFirstChild(bone[1], true)
                local partB = worldModel:FindFirstChild(bone[2], true)
                if partA and partB and partA:IsA("BasePart") and partB:IsA("BasePart") then
                    local posA, visibleA = camera:WorldToViewportPoint(partA.Position)
                    local posB, visibleB = camera:WorldToViewportPoint(partB.Position)
                    if visibleA and visibleB and posA.Z > 0 and posB.Z > 0 then
                        local line = lines[i]
                        line.From = Vector2.new(posA.X, posA.Y)
                        line.To = Vector2.new(posB.X, posB.Y)
                        line.Color = skeletonESP.color
                        line.Thickness = skeletonESP.thickness
                        line.Transparency = 1
                        line.Visible = true
                    end
                end
            end
        end
    end
    for player, lines in pairs(skeletonESP.lines) do
        if player ~= LocalPlayer and not espWorldModels[player] then
            removeSkeletonLines(player)
        end
    end
end)

chamconnections = {
	chams = nil;
	armchams = nil;
	upd = nil;

isArmDescendants = function(part, root)
	local prnt = part.Parent 
	while prnt and prnt ~= root do 
		if prnt.Name == 'Left Arm' or prnt.Name == 'Right Arm' then 
			return true 
		end;
		prnt = prnt.Parent
	end; 
	return false 
end;
doChams = function(w)
	if w.Name ~= 'CurrentWeapon' then 
		return 
	end; 
	for i, v in ipairs(w:GetDescendants()) do 
		if v.Name == 'SurfaceAppearance' then 
			v.Parent = nil;
		elseif v.Name == 'SKIN01' then 
			v.Parent = nil;
		end
		if not v:IsA('BasePart') then 
			continue 
		end;
		
		if v.Name == 'Left Arm' or v.Name == 'Right Arm' then 
			continue 
		end;
		if isArmDescendants(v, w) then 
			continue 
		end;
		v.Material = Enum.Material[FLAGS.world.chamsMaterial]
		v.Color = FLAGS.world.chamsColor
	end;
end;
doArmChams = function(w)
	if w.Name ~= 'CurrentWeapon' then 
		return 
	end;
	for i, v in ipairs(w:GetDescendants()) do 
		if v.Name == 'Left Arm' or v.Name == 'Right Arm' then 
			if v:IsA('BasePart') then 
				v.Material = Enum.Material[FLAGS.world.armChamsMaterial]
				v.Color = FLAGS.world.armChamsColor
			end;
			for _, d in ipairs(v:GetDescendants()) do 
				if d:IsA('BasePart') then
					d.Material = Enum.Material[FLAGS.world.armChamsMaterial]
					d.Color = FLAGS.world.armChamsColor
				end;
				
				if d.Name == 'Shirt' then 
					d.Parent = nil;
				end;
				if d.Name == 'Skin' then 
					if FLAGS.world.viewmodel then 
						d.Transparency = 1
					else
						d.Transparency = 0
					end;
				end;
			end; 
		end;
	end;
end;

updchams = function()
	if chamconnections.upd then 
		chamconnections.upd:Disconnect()
		chamconnections.upd = nil;
	end; 
	if not FLAGS.world.chams and not FLAGS.world.armChams then 
		return 
	end; 
	chamconnections.upd = CurrentCamera.ChildAdded:Connect(function(child)
		if FLAGS.world.chams then 
			doChams(child)
		end;
		if FLAGS.world.armChams then 
			doArmChams(child)
		end;
		task.wait()
		doArmChams(child)
	end)
end;

local bg = LocalPlayer:WaitForChild("PlayerGui"):WaitForChild("GameUI"):WaitForChild("BackgroundUI")
local isEnabled = false

local function setNVFullscreen(enabled)
    local nvUI = bg:FindFirstChild("NightVisionUI")
    if not nvUI then return end

    local inner = nvUI:FindFirstChild("Inner")

    if enabled then
        for _, v in ipairs(nvUI:GetChildren()) do
            if v:IsA("Frame") and v.Name ~= "Inner" then
                v.Visible = false
            end
        end
        if inner then
            local vignette = inner:FindFirstChild("Vignette")
            if vignette then
                vignette.ImageTransparency = 1
                vignette.BackgroundTransparency = 1
            end
            local constraint = inner:FindFirstChildWhichIsA("UIAspectRatioConstraint")
            if constraint then constraint:Destroy() end
            for _, v in ipairs(inner:GetChildren()) do
                if v:IsA("Frame") then
                    v.Visible = false
                end
            end
        end
    else
        for _, v in ipairs(nvUI:GetChildren()) do
            if v:IsA("Frame") then
                v.Visible = true
            end
        end
        if inner then
            local vignette = inner:FindFirstChild("Vignette")
            if vignette then
                vignette.ImageTransparency = 0.2
                vignette.BackgroundTransparency = 1
                if not inner:FindFirstChildWhichIsA("UIAspectRatioConstraint") then
                    local c = Instance.new("UIAspectRatioConstraint")
                    c.AspectRatio = 1.5
                    c.Parent = inner
                end
            end
            for _, v in ipairs(inner:GetChildren()) do
                if v:IsA("Frame") then
                    v.Visible = true
                end
            end
        end
    end
end

bg.ChildAdded:Connect(function(child)
    if child.Name == "NightVisionUI" and isEnabled then
        task.defer(setNVFullscreen, true)
    end
end)

local ItemModuleMap = {}

local pendingItems = {}

local origModuleLoad = BaseItem.ModuleLoad
BaseItem.ModuleLoad = function(module)
    origModuleLoad(module)
    for item in pendingItems do
        if not item.Parent then
            pendingItems[item] = nil
            continue
        end
        local cid = item:GetAttribute("CId")
        if cid and BaseItem.CIDMap[cid] then
            pendingItems[item] = nil
            task.defer(function()
                onGroundItemAdded(item)
            end)
        end
    end
end

local function onGroundItemAdded(item)
    if trackedGroundItems[item] then return end
    if not item.Parent then return end

    local cid = item:GetAttribute("CId")
    if not cid then return end

    local moduleName = BaseItem.CIDMap[cid]
    if not moduleName then
        pendingItems[item] = true
        return
    end

    if not ItemModuleMap[moduleName] then
        local ok, result = pcall(function()
            return gameRequire:GetModuleByName(moduleName)
        end)
        if ok and result then
            ItemModuleMap[moduleName] = result
        end
    end

    local info = ItemModuleMap[moduleName]
    if not info then return end

    local itemType = info.ItemType
    local category

    if itemType == "Weapon" then
        if info.AmmoType then
            category = "weapons"
        else
            category = "melee"
        end
    elseif itemType == "Ammo" then
        category = "ammo"
    elseif itemType == "Axe" or itemType == "Saw" or itemType == "Scissors" then
        category = "melee"
    elseif itemType == "Armor" or itemType == "Gloves" or itemType == "Hat" or itemType == "Mask" or itemType == "Shoe" then
        category = "armor"
    elseif itemType == "Clothing" then
        category = "clothing"
    elseif itemType == "Attachment" then
        category = "attachments"
    elseif itemType == "Backpack" then
        category = "backpacks"
    elseif itemType == "Medical" then
        category = "medical"
    elseif itemType == "Food" or itemType == "Drink" then
        category = "food"
    elseif itemType == "Usable" or itemType == "Consumable" then
        category = "usables"
    elseif itemType == "Resource" or itemType == "Component" or itemType == "Fuel"
        or itemType == "Utility" or itemType == "Compass" or itemType == "Trap" then
        category = "resources"
    else
        return
    end

    local displayName = info.DisplayName or item.Name
    local quantity = item:GetAttribute("Quantity")
    local label = quantity and quantity > 1 and (quantity .. "x " .. displayName) or displayName

    local rootPart = item:IsA("BasePart") and item or item.Parent
    trackedGroundItems[item] = { category, rootPart }
    EspLib.item:add(rootPart, label, category)
end

local function onGroundItemRemoved(item)
    pendingItems[item] = nil
    local data = trackedGroundItems[item]
    if not data then return end
    trackedGroundItems[item] = nil
    EspLib.item:remove(data[2], data[1])
end

function handleWorkspaceChild(child)
    if not child or not child.Parent then
        return
    end
    if not child:IsA("Model") then
        return
    end
    if processedChildren[child] then
        return
    end

    if child.Name == "WorldModel" then
        local states = child:FindFirstChild("States")
        if not states then
            return
        end
        local maxHealth = states:GetAttribute("MaxHealth")
        if not maxHealth then
            return
        end
        if not child:FindFirstChildWhichIsA("BasePart", true) then
            return
        end
        if TrackedVehicles[child] then
            return
        end
        processedChildren[child] = true
        local currentHealth = math.floor((states:GetAttribute("Health") or maxHealth) + 0.5)
        EspLib.item:add(child, "Vehicle [" .. currentHealth .. "/" .. maxHealth .. " HP]", "vehicles")
        TrackedVehicles[child] = { states = states, maxHealth = maxHealth }
        return
    end

    if child.Name == "HeliModel" then
        if not child:FindFirstChildWhichIsA("BasePart", true) then
            return
        end
        if TrackedHelicopters[child] then
            return
        end
        processedChildren[child] = true
        local health = math.floor((child:GetAttribute("Health") or 1250) + 0.5)
        EspLib.item:add(child, "Helicopter [" .. health .. "/1250 HP]", "helis")
        TrackedHelicopters[child] = true
        return
    end

    local trapLabel = TrapModelNames[child.Name]
    if trapLabel then
        if not child:FindFirstChildWhichIsA("BasePart", true) then
            return
        end
        processedChildren[child] = true
        EspLib.item:add(child, trapLabel, "traps")
        return
    end

    local airdropLabel = AirdropModelNames[child.Name]
    if airdropLabel then
        if not child:FindFirstChildWhichIsA("BasePart", true) then
            return
        end
        processedChildren[child] = true
        EspLib.item:add(child, airdropLabel, "airdrops")
    end
end

-- Merged RenderStepped
Services.RunService.RenderStepped:Connect(LPH_NO_VIRTUALIZE(function(dt)
    -- FOV circle
    if FLAGS.aimbot.fov_circle then
        local vp = CurrentCamera.ViewportSize
        local deg = math.clamp(FLAGS.aimbot.fov_size or 90, 0.5, 179.5)
        local r = math.tan(math.rad(deg / 2)) / math.tan(math.rad(CurrentCamera.FieldOfView / 2)) * (vp.Y / 2)
        local cx, cy = vp.X / 2, vp.Y / 2
        if FLAGS.aimbot.fov_gradient then
            fov_spin = (fov_spin + dt * 90) % 360
        end
        for i = 1, fov_segments do
            local a1 = math.rad((i - 1) / fov_segments * 360 + fov_spin)
            local a2 = math.rad(i / fov_segments * 360 + fov_spin)
            local c
            if FLAGS.aimbot.fov_gradient then
                local t = ((i - 1) / fov_segments + 0.5) % 1
                local t2 = t < 0.5 and t * 2 or 1 - (t - 0.5) * 2
                c = (FLAGS.aimbot.fov_color_a or Color3.new(1,0,0)):Lerp(FLAGS.aimbot.fov_color_b or Color3.new(0,0,1), t2)
            else
                c = FLAGS.aimbot.fov_color or Color3.new(1, 1, 1)
            end
            local l = fov_lines[i]
            l.From = Vector2.new(cx + math.cos(a1) * r, cy + math.sin(a1) * r)
            l.To   = Vector2.new(cx + math.cos(a2) * r, cy + math.sin(a2) * r)
            l.Color = c
            l.Thickness = FLAGS.aimbot.fov_thickness or 1
            l.Visible = true
        end
    else
        fov_spin = 0
        for i = 1, fov_segments do
            fov_lines[i].Visible = false
        end
    end

    -- snaplines
    do
        if FLAGS.visuals.snaplines then
            local target = GetClosestPlayer()
            if target and target.TargetPart then
                local screenPos, onScreen = CurrentCamera:WorldToViewportPoint(target.TargetPart.Position)
                if onScreen then
                    local vp = CurrentCamera.ViewportSize
                    snapline.From = Vector2.new(vp.X / 2, vp.Y)
                    snapline.To = Vector2.new(screenPos.X, screenPos.Y)

                    snapline.Thickness = FLAGS.visuals.snaplines_thickness
                    snapline.Visible = true
                else
                    snapline.Visible = false
                end
            else
                snapline.Visible = false
            end
        else
            snapline.Visible = false
        end
    end

    -- antiaim yaw apply + pitch
    if FLAGS.aimbot.antiaim.enabled and not (FLAGS.aimbot.antiaim.noladderinfluence and (isClimbing and isClimbing())) then
        if FLAGS.aimbot.antiaim.yaw_mode == "Spin" then
            yaw = FLAGS.aimbot.antiaim.spin_angle
        elseif FLAGS.aimbot.antiaim.yaw_mode == "Forwards" then
            yaw = get_cam_yaw()
        elseif FLAGS.aimbot.antiaim.yaw_mode == "Backwards" then
            yaw = get_cam_yaw() + math.pi
        elseif FLAGS.aimbot.antiaim.yaw_mode == "Jitter" then
            yaw = (get_cam_yaw() - math.pi) + FLAGS.aimbot.antiaim.yaw_jitter_offset
        elseif FLAGS.aimbot.antiaim.yaw_mode == "Skitter" then
            yaw = (get_cam_yaw() - math.pi) + FLAGS.aimbot.antiaim.yaw_jitter_offset
        end
        if yaw then
            Modules.char_plugin:SetYaw(yaw)
            if LocalPlayer.CharData and LocalPlayer.CharData:FindFirstChild("Yaw") then
                LocalPlayer.CharData.Yaw.Value = yaw
            end
            local world_char = Modules.CustomMeshCharacter:GetWorldCharacterFromPlayer(LocalPlayer)
            if world_char and world_char:FindFirstChild("HumanoidRootPart") then
                pcall(function()
                    world_char.HumanoidRootPart.CFrame = CFrame.new(world_char.HumanoidRootPart.Position) * CFrame.Angles(0, yaw, 0)
                end)
            end
        end
    end
end))

local get_closest_zombie = function()
    local mouselocation = UserInputService:GetMouseLocation()
    local closest = nil
    local closestdist = math.huge
    for _, v in entities:GetChildren() do
        if v:FindFirstChild("UpperTorso") then
            continue
        end
        local hrp = v:FindFirstChild("HumanoidRootPart")
        if not hrp then
            continue
        end
        if hrp.CollisionGroup == "PropGroup" then
            continue
        end
		local head = v:FindFirstChild("Head")
		
		if not head then 
			continue 
		end 
        local screenpos = CurrentCamera:WorldToViewportPoint(hrp.Position)
        if screenpos.Z < 0 then
            continue
        end
        local dist = (Vector2.new(screenpos.X, screenpos.Y) - mouselocation).Magnitude
        if dist < closestdist then
            closestdist = dist
            closest = v
        end
    end
    return closest
end
local get_closest_player_world = function()
    local mouselocation = UserInputService:GetMouseLocation()
    local closest = nil
    local closestdist = math.huge
    for _, data in Modules.CustomMeshCharacter:GetCharacters() do
        local player = data.Player
        if player == LocalPlayer or IsDead(player) or IsInSquad(player) then
            continue
        end
        local hrp = data.Character and data.Character:FindFirstChild("Part")
        if not hrp then
            continue
        end
        local screenpos = CurrentCamera:WorldToViewportPoint(hrp.Position)
        if screenpos.Z < 0 then
            continue
        end
        local dist = (Vector2.new(screenpos.X, screenpos.Y) - mouselocation).Magnitude
        if dist < closestdist then
            closestdist = dist
            closest = data
        end
    end
    return closest
end
local get_killaura_target = function()
    local targets = FLAGS.aimbot.killaura_targets
    local mouseloc = UserInputService:GetMouseLocation()
    local best = nil
    local bestdist = math.huge
    local besttype = nil
    local function checktarget(targettype, pos, data)
        local screenpos = CurrentCamera:WorldToViewportPoint(pos)
        if screenpos.Z < 0 then
            return
        end
        local dist = (Vector2.new(screenpos.X, screenpos.Y) - mouseloc).Magnitude
        if dist < bestdist then
            bestdist = dist
            best = data
            besttype = targettype
        end
    end
    if table.find(targets, "zombies") then
        local z = get_closest_zombie()
        if z then
            checktarget("zombie", z.HumanoidRootPart.Position, z)
        end
    end
    if table.find(targets, "players") then
        local p = get_closest_player_world()
        if p then
            local hrp = p.Character:FindFirstChild("Part")
            if hrp then
                checktarget("player", hrp.Position, p)
            end
        end
    end
    return best, besttype
end

local instantAimHooks = {}
local function setInstantAim(state)
	FLAGS.aimbot.exploits.instant_aim = state
	if state then
		for _, v in next, getgc(true) do
			if type(v) == 'table' then
				for i, val in next, v do
					if type(i) == 'string' and i:find('GunAim') then
						if type(val) == 'number' then
							instantAimHooks[v] = instantAimHooks[v] or {}
							instantAimHooks[v][i] = val
							v[i] = 100000000
						elseif type(val) == 'function' then
							if not isfunctionhooked(val) then
								local old; old = hookfunction(val, function()
									return 100000000
								end)
								instantAimHooks[val] = old
							end
						end
					end
				end
			end
		end
	else
		-- restore number values
		for tbl, fields in pairs(instantAimHooks) do
			if type(tbl) == 'table' then
				for k, orig in pairs(fields) do
					tbl[k] = orig
				end
			elseif type(tbl) == 'function' then
				if isfunctionhooked(tbl) then
					restorefunction(tbl)
				end
			end
		end
		table.clear(instantAimHooks)
	end
end

--#region Hitbox Expander
local function SetHeads(Character, Remove)
	for _, Head in Character:QueryDescendants("[$FakeHead]") do
		Head:Destroy()
	end
	if not FLAGS.aimbot.exploits.hitbox_expander or Remove then
		return
	end
	local Radius = FLAGS.aimbot.exploits.hitbox_radius
	local RealHead = Character:FindFirstChild("Head")
	if not RealHead then return end

	local RealNeck = RealHead:FindFirstChild("Neck")
	local BaseC0 = RealNeck and RealNeck.C0 or CFrame.identity

	local VisualHead = RealHead:Clone()
	pcall(function() VisualHead.face:Destroy() end)
	VisualHead.Shape = Enum.PartType.Ball
	VisualHead.Size = Vector3.one * Radius * 2
	VisualHead.Color = Color3.fromRGB(255, 50, 50)
	VisualHead.Material = Enum.Material.ForceField
	VisualHead.CanCollide = false
	VisualHead.CanQuery = false
	VisualHead.CanTouch = false
	VisualHead.Massless = true
	VisualHead.CastShadow = false
	VisualHead:SetAttribute("FakeHead", true)
	VisualHead.Transparency = 0.5
	VisualHead.Parent = Character

	local SearchRadius = Radius - 0.5
	for X = -SearchRadius, SearchRadius do
		for Y = -SearchRadius, SearchRadius do
			for Z = -SearchRadius, SearchRadius do
				local Distance = math.sqrt(X*X + Y*Y + Z*Z) - SearchRadius
				if Distance <= 0.5 and Distance >= -0.5 then
					local NewHead = RealHead:Clone()
					pcall(function() NewHead.face:Destroy() end)
					NewHead.CanCollide = true
					NewHead.CanQuery = true
					NewHead.CanTouch = false
					NewHead.Massless = true
					NewHead.CastShadow = false
					NewHead.Transparency = 1
					NewHead:SetAttribute("FakeHead", true)
					NewHead.Parent = Character
					local NeckJoint = NewHead:FindFirstChild("Neck")
					if NeckJoint then
						NeckJoint.C0 = BaseC0 * CFrame.new(X, Y, Z)
					end
				end
			end
		end
	end
end

local hitboxCharAddedConn = nil
local hitboxCharRemovedConn = nil

local function applyHitboxExpander(enabled)
	FLAGS.aimbot.exploits.hitbox_expander = enabled

	-- disconnect old connections
	if hitboxCharAddedConn then hitboxCharAddedConn:Disconnect(); hitboxCharAddedConn = nil end
	if hitboxCharRemovedConn then hitboxCharRemovedConn:Disconnect(); hitboxCharRemovedConn = nil end

	-- apply/remove on existing characters
	for _, Data in Modules.CustomMeshCharacter:GetCharacters() do
		if Data.Player == LocalPlayer then continue end
		SetHeads(Data.WorldModel, not enabled)
	end

	if not enabled then return end

	hitboxCharAddedConn = Modules.CustomMeshCharacter.CharacterAdded:Connect(function(Player, Character, WorldModel)
		if Player == LocalPlayer then return end
		SetHeads(WorldModel)
	end)
	hitboxCharRemovedConn = Modules.CustomMeshCharacter.CharacterRemoved:Connect(function(Player, Character, WorldModel)
		SetHeads(WorldModel, true)
	end)
end

							local totalHeight=(height*0.8)+targetY
							local worldVertical=Vector3.new(0,totalHeight,0)
							local convertedVertical=torso.CFrame:VectorToObjectSpace(worldVertical)
							local localOffset=Vector3.new(targetX,0,targetZ)+convertedVertical
							neck.C0=defaultC0+localOffset
							head.CFrame=torso.CFrame*neck.C0
							head.Size=Vector3.new(1,1,1)

						end
					end
				end

	do
		local RunService = game:GetService("RunService")
		local UserInputService = game:GetService("UserInputService")
		local drawings = {}
		for i = 1, 16 do
			local ok, obj = pcall(function() return Drawing.new("Line") end)
			if ok and obj then drawings[#drawings + 1] = obj end
		end
		local dot
		pcall(function() dot = Drawing.new("Circle"); dot.Filled = true; dot.NumSides = 24; dot.Visible = false end)
		local rotation = 0
		local function drawGradientSegment(index, fromPos, toPos, colorA, colorB, thickness)
			local lineA, lineB = drawings[index], drawings[index + 1]
			local mid = (fromPos + toPos) / 2
			if lineA then
				lineA.From = fromPos; lineA.To = mid; lineA.Color = colorA
				lineA.Thickness = thickness; lineA.Visible = true
			end
			if lineB then
				lineB.From = mid; lineB.To = toPos
				lineB.Color = crossCfg.gradient and colorB or colorA
				lineB.Thickness = thickness; lineB.Visible = true
			end
		end
		RunService.RenderStepped:Connect(function(dt)
			local ok = pcall(function()
				if not crossCfg.enabled then
					for _, line in ipairs(drawings) do line.Visible = false end
					if dot then dot.Visible = false end
					return
				end
				rotation = (rotation + crossCfg.spin_speed * dt) % 360
				local pos = UserInputService:GetMouseLocation()
				for _, line in ipairs(drawings) do line.Visible = false end
				if dot then dot.Visible = false end
				if crossCfg.style == "Dot" and dot then
					dot.Position = pos; dot.Radius = math.max(1, crossCfg.thickness); dot.Color = crossCfg.color; dot.Filled = true; dot.Visible = true
				elseif crossCfg.style == "Circle" and dot then
					dot.Position = pos; dot.Radius = crossCfg.size; dot.Thickness = crossCfg.thickness; dot.Color = crossCfg.color; dot.Filled = false; dot.Visible = true
				elseif crossCfg.style == "Swastika" then
					for arm = 1, 4 do
						local a = math.rad(rotation) + (arm - 1) * math.pi / 2
						local d = Vector2.new(math.cos(a), math.sin(a))
						local bend = Vector2.new(math.cos(a + math.pi / 2), math.sin(a + math.pi / 2))
						local startPos = pos + d * crossCfg.gap
						local elbow = pos + d * (crossCfg.gap + crossCfg.size)
						local endPos = elbow + bend * (crossCfg.size * 0.65)
						local idx = (arm - 1) * 4 + 1
						drawGradientSegment(idx, startPos, elbow, crossCfg.color, crossCfg.color2, crossCfg.thickness)
						drawGradientSegment(idx + 2, elbow, endPos, crossCfg.color, crossCfg.color2, crossCfg.thickness)
					end
				else
					local r = math.rad(rotation)
					local dirs = {{0,-1},{0,1},{-1,0},{1,0}}
					for i, d in ipairs(dirs) do
						local x = d[1] * math.cos(r) - d[2] * math.sin(r)
						local y = d[1] * math.sin(r) + d[2] * math.cos(r)
						local fromPos = pos + Vector2.new(x, y) * crossCfg.gap
						local toPos = pos + Vector2.new(x, y) * (crossCfg.gap + crossCfg.size)
						drawGradientSegment((i - 1) * 2 + 1, fromPos, toPos, crossCfg.color, crossCfg.color2, crossCfg.thickness)
					end
				end
			end)
			if not ok then
				for _, line in ipairs(drawings) do pcall(function() line.Visible = false end) end
				if dot then pcall(function() dot.Visible = false end) end
			end
		end)
	end

	local function setNotificationPosition(position)
		if position ~= "Top Left" and position ~= "Top Right" then position = "Top Right" end
		FLAGS.misc.notification_position = position
		pcall(function()
			local holder = Library.NotifHolder and Library.NotifHolder.Instance
			local layout = holder and holder:FindFirstChildOfClass("UIListLayout")
			if not holder or not layout then return end
			layout.VerticalAlignment = Enum.VerticalAlignment.Top
			if position == "Top Left" then
				holder.AnchorPoint = Vector2.new(0, 0)
				-- Keep notifications below the Roblox top bar/logo with a small left margin.
				holder.Position = UDim2.new(0, 8, 0, 48)
				layout.HorizontalAlignment = Enum.HorizontalAlignment.Left
			else
				holder.AnchorPoint = Vector2.new(1, 0)
				holder.Position = UDim2.new(1, 0, 0, 0)
				layout.HorizontalAlignment = Enum.HorizontalAlignment.Right
			end
		end)
	end
